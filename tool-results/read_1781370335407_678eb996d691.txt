   202→export default function ShippingPage() {
   203→  
   204→  const confirmAction = useConfirm();
   205→  const [shipments, setShipments] = useState([]);
   206→  const [carriers, setCarriers] = useState([]);
   207→  const [config, setConfig] = useState({ api_key: "", api_key_set: false, default_carrier: "aramex", auto_track: true, connected: false });
   208→  const [filter, setFilter] = useState("all");
   209→  const [search, setSearch] = useState("");
   210→  const [loading, setLoading] = useState(true);
   211→  const [refreshing, setRefreshing] = useState(null);
   212→  const [sendingUpdate, setSendingUpdate] = useState(null);
   213→
   214→  // Config panel state
   215→  const [showApiKey, setShowApiKey] = useState(false);
   216→  const [configForm, setConfigForm] = useState({ api_key: "", default_carrier: "aramex", auto_track: true });
   217→  const [savingConfig, setSavingConfig] = useState(false);
   218→
   219→  // Add tracking modal state
   220→  const [showAddModal, setShowAddModal] = useState(false);
   221→  const [addForm, setAddForm] = useState({ order_id: "", tracking_number: "", carrier: "aramex", title: "" });
   222→  const [addingTracking, setAddingTracking] = useState(false);
   223→  const [orders, setOrders] = useState([]);
   224→
   225→  // Timeline panel state
   226→  const [viewingShipment, setViewingShipment] = useState(null);
   227→
   228→  // Demo mode
   229→  const [isDemo, setIsDemo] = useState(false);
   230→
   231→  const supabase = createClient();
   232→  const { currentStoreId } = useCurrentStore();
   233→  const toast = useToast();
   234→
   235→  // Fetch config
   236→  const fetchConfig = useCallback(async () => {
   237→    try {
   238→      const res = await fetch("/api/shipping/config");
   239→      if (res.ok) {
   240→        const data = await res.json();
   241→        if (data.success) {
   242→          setConfig(data.config);
   243→          setConfigForm({
   244→            api_key: "",
   245→            default_carrier: data.config.default_carrier || "aramex",
   246→            auto_track: data.config.auto_track !== false,
   247→          });
   248→          setIsDemo(!data.config.connected);
   249→        }
   250→      }
   251→    } catch (err) {
   252→      console.error("Failed to fetch config:", err);
   253→    }
   254→  }, []);
   255→
   256→  // Fetch carriers
   257→  const fetchCarriers = useCallback(async () => {
   258→    try {
   259→      const res = await fetch("/api/shipping/carriers");
   260→      if (res.ok) {
   261→        const data = await res.json();
   262→        if (data.success) setCarriers(data.carriers || []);
   263→      }
   264→    } catch (err) {
   265→      console.error("Failed to fetch carriers:", err);
   266→    }
   267→  }, []);
   268→
   269→  // Fetch shipments
   270→  const fetchShipments = useCallback(async () => {
   271→    setLoading(true);
   272→    try {
   273→      const { data: { user } } = await supabase.auth.getUser();
   274→      let query = supabase
   275→        .from("shipment_trackings")
   276→        .select("*")
   277→        .order("created_at", { ascending: false });
   278→      
   279→      if (user) query = query.eq("account_id", user.id);
   280→
   281→      const { data, error } = await query;
   282→
   283→      if (!error && data) {
   284→        setShipments(data);
   285→        // If no shipments and no API key, use demo data
   286→        if (data.length === 0 && !config.connected) {
   287→          setIsDemo(true);
   288→          setShipments(DEMO_SHIPMENTS);
   289→        }
   290→      } else if (!config.connected) {
   291→        setIsDemo(true);
   292→        setShipments(DEMO_SHIPMENTS);
   293→      }
   294→    } catch (err) {
   295→      if (!config.connected) {
   296→        setIsDemo(true);
   297→        setShipments(DEMO_SHIPMENTS);
   298→      }
   299→    }
   300→    setLoading(false);
   301→  }, [config.connected, currentStoreId]);
   302→
   303→  // Fetch orders for dropdown
   304→  const fetchOrders = useCallback(async () => {
   305→    try {
   306→      const { data: { user } } = await supabase.auth.getUser();
   307→      if (!user) return;
   308→      let query = supabase
   309→        .from("orders")
   310→        .select("id, order_number, status")
   311→        .eq("account_id", user.id)
   312→        .order("created_at", { ascending: false })
   313→        .limit(20);
   314→      if (currentStoreId) query = query.or(`store_id.eq.${currentStoreId},store_id.is.null`);
   315→      const { data } = await query;
   316→      setOrders(data || []);
   317→    } catch (err) {
   318→      console.error("Failed to fetch orders:", err);
   319→    }
   320→  }, [currentStoreId]);
   321→
   322→  useEffect(() => { fetchConfig(); fetchCarriers(); }, []);
   323→  useEffect(() => { fetchShipments(); }, [fetchShipments]);
   324→  useEffect(() => { fetchOrders(); }, [fetchOrders]);
   325→
   326→  // Save config
   327→  const handleSaveConfig = async () => {
   328→    setSavingConfig(true);
   329→    try {
   330→      const payload = {
   331→        default_carrier: configForm.default_carrier,
   332→        auto_track: configForm.auto_track,
   333→      };
   334→      // Only send API key if user entered one
   335→      if (configForm.api_key.trim()) {
   336→        payload.api_key = configForm.api_key.trim();
   337→      }
   338→      const res = await fetch("/api/shipping/config", {
   339→        method: "PUT",
   340→        headers: { "Content-Type": "application/json" },
   341→        body: JSON.stringify(payload),
   342→      });
   343→      const data = await res.json();
   344→      if (data.success) {
   345→        await fetchConfig();
   346→        await fetchShipments();
   347→        setConfigForm((prev) => ({ ...prev, api_key: "" }));
   348→      } else {
   349→        toast.error("Failed to save config: " + (data.error || "Unknown error"));
   350→      }
   351→    } catch (err) {
   352→      toast.error("Error: " + err.message);
   353→    }
   354→    setSavingConfig(false);
   355→  };
   356→
   357→  // Add tracking
   358→  const handleAddTracking = async () => {
   359→    if (!addForm.tracking_number.trim()) {
   360→      toast.warning("Tracking number is required");
   361→      return;
   362→    }
   363→    setAddingTracking(true);
   364→    try {
   365→      const res = await fetch("/api/shipping/track", {
   366→        method: "POST",
   367→        headers: { "Content-Type": "application/json" },
   368→        body: JSON.stringify(addForm),
   369→      });
   370→      const data = await res.json();
   371→      if (data.success) {
   372→        setShowAddModal(false);
   373→        setAddForm({ order_id: "", tracking_number: "", carrier: "aramex", title: "" });
   374→        await fetchShipments();
   375→      } else {
   376→        toast.error("Failed to add tracking: " + (data.error || "Unknown error"));
   377→      }
   378→    } catch (err) {
   379→      toast.error("Error: " + err.message);
   380→    }
   381→    setAddingTracking(false);
   382→  };
   383→
   384→  // Refresh tracking
   385→  const handleRefresh = async (shipment) => {
   386→    if (shipment.id?.startsWith("demo-")) {
   387→      // Simulate refresh for demo
   388→      setRefreshing(shipment.id);
   389→      setTimeout(() => setRefreshing(null), 1000);
   390→      return;
   391→    }
   392→    setRefreshing(shipment.id);
   393→    try {
   394→      const res = await fetch("/api/shipping/track", {
   395→        method: "PATCH",
   396→        headers: { "Content-Type": "application/json" },
   397→        body: JSON.stringify({ id: shipment.id }),
   398→      });
   399→      const data = await res.json();
   400→      if (data.success) {
   401→        await fetchShipments();
   402→        // Update viewing shipment if it's the same
   403→        if (viewingShipment?.id === shipment.id) {
   404→          setViewingShipment(data.tracking);
   405→        }
   406→      }
   407→    } catch (err) {
   408→      console.error("Refresh error:", err);
   409→    }
   410→    setRefreshing(null);
   411→  };
   412→
   413→  // Send update to customer
   414→  const handleSendUpdate = async (shipment) => {
   415→    setSendingUpdate(shipment.id);
   416→    try {
   417→      const statusInfo = STATUS_CONFIG[shipment.status] || STATUS_CONFIG.pending;
   418→      const message = `📦 Shipment Update\n\nTracking: ${shipment.tracking_number}\nCarrier: ${shipment.carrier}\nStatus: ${statusInfo.label}\n${shipment.estimated_delivery ? "Est. Delivery: " + formatDate(shipment.estimated_delivery) : ""}`;
   419→
   420→      // Find the order's customer conversation
   421→      if (shipment.order_id) {
   422→        const { data: order } = await supabase
   423→          .from("orders")
   424→          .select("customer_id, order_number")
   425→          .eq("id", shipment.order_id)
   426→          .single();
   427→
   428→        if (order?.customer_id) {
   429→          const { data: conv } = await supabase
   430→            .from("conversations")
   431→            .select("id")
   432→            .eq("customer_id", order.customer_id)
   433→            .eq("account_id", (await supabase.auth.getUser()).data.user?.id)
   434→            .limit(1)
   435→            .single();
   436→
   437→          if (conv) {
   438→            await fetch("/api/messages/send", {
   439→              method: "POST",
   440→              headers: { "Content-Type": "application/json" },
   441→              body: JSON.stringify({
   442→                conversationId: conv.id,
   443→                content: message,
   444→                type: "text",
   445→              }),
   446→            });
   447→            toast.success("Shipping update sent to customer!");
   448→          } else {
   449→            toast.warning("No conversation found for this order's customer.");
   450→          }
   451→        } else {
   452→          toast.warning("No customer associated with this order.");
   453→        }
   454→      } else {
   455→        toast.warning("This tracking is not linked to an order with a customer.");
   456→      }
   457→    } catch (err) {
   458→      toast.error("Error sending update: " + err.message);
   459→    }
   460→    setSendingUpdate(null);
   461→  };
   462→
   463→  // Delete tracking
   464→  const handleDelete = async (shipment) => {
   465→    if (!(await confirmAction("Delete this tracking record?"))) return;
   466→    if (shipment.id?.startsWith("demo-")) return;
   467→    try {
   468→      await fetch(`/api/shipping/track?id=${shipment.id}`, { method: "DELETE" });
   469→      await fetchShipments();
   470→      if (viewingShipment?.id === shipment.id) setViewingShipment(null);
   471→    } catch (err) {
   472→      console.error("Delete error:", err);
   473→    }
   474→  };
   475→
   476→  // Format helpers
   477→  const formatDate = (d) => {
   478→    if (!d) return "—";
   479→    return new Date(d).toLocaleDateString("en-US", { month: "short", day: "numeric", year: "numeric" });
   480→  };
   481→
   482→  const formatDateTime = (d) => {
   483→    if (!d) return "—";
   484→    return new Date(d).toLocaleDateString("en-US", { month: "short", day: "numeric", hour: "2-digit", minute: "2-digit" });
   485→  };
   486→
   487→  const formatTimeAgo = (d) => {
   488→    if (!d) return "";
   489→    const diff = Date.now() - new Date(d).getTime();
   490→    const mins = Math.floor(diff / 60000);
   491→    if (mins < 60) return `${mins}m ago`;
   492→    const hours = Math.floor(mins / 60);
   493→    if (hours < 24) return `${hours}h ago`;
   494→    const days = Math.floor(hours / 24);
   495→    return `${days}d ago`;
   496→  };
   497→
   498→  // Filtered shipments
   499→  const filteredShipments = shipments.filter((s) => {
   500→    if (filter !== "all" && s.status !== filter) return false;
   501→    if (search) {
   502→      const q = search.toLowerCase();
   503→      return (
   504→        s.tracking_number?.toLowerCase().includes(q) ||
   505→        s.title?.toLowerCase().includes(q) ||
   506→        s.carrier?.toLowerCase().includes(q) ||
   507→        s.order_id?.toLowerCase().includes(q)
   508→      );
   509→    }
   510→    return true;
   511→  });
   512→
   513→  // Stats
   514→  const stats = {
   515→    total: shipments.length,
   516→    in_transit: shipments.filter((s) => s.status === "in_transit").length,
   517→    delivered: shipments.filter((s) => s.status === "delivered").length,
   518→    exception: shipments.filter((s) => s.status === "exception").length,
   519→  };
   520→
   521→  return (
   522→    <>
   523→      {/* Demo Mode Banner */}
   524→      {isDemo && (
   525→        <div className="shipping-demo-banner">
   526→          <div className="shipping-demo-banner-content">
   527→            <Info size={16} />
   528→            <span>
   529→              Running in <strong>demo mode</strong>. Connect AfterShip for live tracking data.
   530→            </span>
   531→          </div>
   532→          <button
   533→            className="btn btn-sm"
   534→            style={{ background: "rgba(0, 210, 255, 0.15)", color: "var(--accent-secondary)", border: "1px solid rgba(0, 210, 255, 0.3)" }}
   535→            onClick={() => document.getElementById("shipping-config-section")?.scrollIntoView({ behavior: "smooth" })}
   536→          >
   537→            Connect Now
   538→          </button>
   539→        </div>
   540→      )}
   541→
   542→      <div className="page-header">
   543→        <h1>Shipping</h1>
   544→        <div className="page-header-actions">
   545→          <button className="btn btn-primary" onClick={() => setShowAddModal(true)}>
   546→            <Plus size={16} />
   547→            Add Tracking
   548→          </button>
   549→        </div>
   550→      </div>
   551→
   552→      {/* Stats */}
   553→      <div className="stats-grid" style={{ marginBottom: "var(--space-lg)" }}>
   554→        <div className="stat-card">
   555→          <div className="stat-card-header">
   556→            <div className="stat-card-icon purple"><Truck size={20} /></div>
   557→          </div>
   558→          <div className="stat-card-value">{stats.total}</div>
   559→          <div className="stat-card-label">Total Shipments</div>
   560→        </div>
   561→        <div className="stat-card">
   562→          <div className="stat-card-header">
   563→            <div className="stat-card-icon blue"><Package size={20} /></div>
   564→          </div>
   565→          <div className="stat-card-value">{stats.in_transit}</div>
   566→          <div className="stat-card-label">In Transit</div>
