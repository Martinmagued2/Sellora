   132→   132→    if (!activeCoupons.length) return null;

   134→   134→    let bestSavings = 0;
   135→   135→
   136→   136→    for (const coupon of activeCoupons) {
   137→   137→      // Check if coupon applies to this product
   138→   138→      let applies = false;
   139→   139→      if (coupon.applies_to === "all") {
   140→   140→        applies = true;
   141→   141→      } else if (coupon.applies_to === "specific_products") {
   142→   142→        applies = Array.isArray(coupon.product_ids) && coupon.product_ids.includes(product.id);
   143→   143→      } else if (coupon.applies_to === "specific_categories") {
   144→   144→        applies = Array.isArray(coupon.categories) && product.category && coupon.categories.includes(product.category);
   145→   145→      }
   146→   146→
   147→   147→      if (!applies) continue;
   148→   148→
   149→   149→      // Check min_order_value (product price as proxy)
   150→   150→      if (coupon.min_order_value && product.price < coupon.min_order_value) continue;
   151→   151→
   152→   152→      // Calculate savings
   153→   153→      let savings = 0;
   154→   154→      if (coupon.type === "percentage") {
   155→   155→        savings = product.price * (coupon.value / 100);
   156→   156→      } else if (coupon.type === "fixed") {
   157→   157→        savings = Math.min(parseFloat(coupon.value), product.price);
   158→   158→      } else if (coupon.type === "free_shipping") {
   159→   159→        // Free shipping doesn't change product price display
   160→   160→        continue;
   161→   161→      }
   162→   162→
   163→   163→      // Pick the coupon that gives the biggest discount
   164→   164→      if (savings > bestSavings) {
   165→   165→        bestSavings = savings;
   166→   166→        bestDiscount = {
   167→   167→          coupon,
   168→   168→          discountAmount: savings,
   169→   169→          newPrice: Math.max(product.price - savings, 0),
   170→   170→          label: coupon.type === "percentage" ? `${coupon.value}% OFF` : `${parseFloat(coupon.value).toLocaleString()} EGP OFF`,
   171→   171→        };
   172→   172→      }
   173→   173→    }
   174→   174→
   175→   175→    return bestDiscount;
