    38→  const [editingProduct, setEditingProduct] = useState(null);
    39→  const [imageFile, setImageFile] = useState(null);
    40→  const [imagePreview, setImagePreview] = useState(null);
    41→  const [dragOver, setDragOver] = useState(false);
    42→  const [accountPlan, setAccountPlan] = useState("starter");
    43→  const [generatingImage, setGeneratingImage] = useState(false);

    45→  const [generatedImageUrl, setGeneratedImageUrl] = useState(null);
    46→  const [generatingDesc, setGeneratingDesc] = useState(false);
    47→  const [aiDescEnglish, setAiDescEnglish] = useState("");
    48→  const [aiDescArabic, setAiDescArabic] = useState("");
    49→  const [aiPriceSuggestion, setAiPriceSuggestion] = useState("");
    50→  const [topRecommended, setTopRecommended] = useState([]);
    51→  const [recLoading, setRecLoading] = useState(false);
    52→
    53→  // Variants state
    54→  const [variants, setVariants] = useState([]);
    55→
    56→  // Active coupons for discount display

    59→  // View product details modal
    60→  const [viewProduct, setViewProduct] = useState(null);
    61→
    62→  const fileInputRef = useRef(null);

    69→
    70→  // Ensure storage buckets exist

    75→      console.warn("[Products] Bucket ensure failed:", e.message);

    78→
    79→  const fetchProducts = useCallback(async () => {
    80→    setLoading(true);
    81→
    82→    const { data: { user } } = await supabase.auth.getUser();
    83→    if (user) {
    84→      const { data: account } = await supabase.from("accounts").select("plan").eq("id", user.id).single();

    89→    let query = supabase.from("products").select("*").eq("account_id", user.id).order("created_at", { ascending: false });

    91→    if (filter === "active") query = query.eq("status", "active");

   103→
   104→  // Fetch active coupons for discount display
   105→  useEffect(() => {
   106→    const fetchActiveCoupons = async () => {
   107→      try {
   108→        const { data: { user } } = await supabase.auth.getUser();
   109→        if (!user) return;
   110→        const now = new Date().toISOString();
   111→        const { data, error } = await supabase
   112→          .from("coupons")
   113→          .select("*")
   114→          .eq("account_id", user.id)
   115→          .eq("is_active", true)
   116→          .lte("starts_at", now)
   117→          .or(`expires_at.is.null,expires_at.gte.${now}`);
   118→        if (!error && data) {
   119→          // Filter out exhausted coupons
   120→          const active = data.filter(c => c.max_uses === null || c.used_count < c.max_uses);
   121→          setActiveCoupons(active);
   122→        }
   123→      } catch (err) {
   124→        console.warn("Failed to fetch active coupons:", err.message);
   125→      }
   126→    };
   127→    fetchActiveCoupons();
   128→  }, [supabase]);
   129→
   130→  // Helper: find the best applicable coupon for a product
   131→  const getDiscountForProduct = useCallback((product) => {
   132→    if (!activeCoupons.length) return null;
   133→    let bestDiscount = null;
   134→    let bestSavings = 0;
   135→
   136→    for (const coupon of activeCoupons) {
   137→      // Check if coupon applies to this product
   138→      let applies = false;
   139→      if (coupon.applies_to === "all") {
   140→        applies = true;
   141→      } else if (coupon.applies_to === "specific_products") {
   142→        applies = Array.isArray(coupon.product_ids) && coupon.product_ids.includes(product.id);
   143→      } else if (coupon.applies_to === "specific_categories") {
   144→        applies = Array.isArray(coupon.categories) && product.category && coupon.categories.includes(product.category);
   145→      }
   146→
   147→      if (!applies) continue;
   148→
   149→      // Check min_order_value (product price as proxy)
   150→      if (coupon.min_order_value && product.price < coupon.min_order_value) continue;
   151→
   152→      // Calculate savings
   153→      let savings = 0;
   154→      if (coupon.type === "percentage") {
   155→        savings = product.price * (coupon.value / 100);
   156→      } else if (coupon.type === "fixed") {
   157→        savings = Math.min(parseFloat(coupon.value), product.price);
   158→      } else if (coupon.type === "free_shipping") {
   159→        // Free shipping doesn't change product price display
   160→        continue;
   161→      }
   162→
   163→      // Pick the coupon that gives the biggest discount
   164→      if (savings > bestSavings) {
   165→        bestSavings = savings;
   166→        bestDiscount = {
   167→          coupon,
   168→          discountAmount: savings,
   169→          newPrice: Math.max(product.price - savings, 0),
   170→          label: coupon.type === "percentage" ? `${coupon.value}% OFF` : `${parseFloat(coupon.value).toLocaleString()} EGP OFF`,
   171→        };
   172→      }
   173→    }
   174→
   175→    return bestDiscount;
   176→  }, [activeCoupons]);
   177→
   178→  // Fetch top recommended products
   179→  const fetchTopRecommended = useCallback(async () => {
   180→    setRecLoading(true);
   181→    try {
   182→      const { data: { user } } = await supabase.auth.getUser();
   183→      if (!user) return;
   184→
   185→      const { data: orders } = await supabase
   186→        .from("orders")
   187→        .select("items")
   188→        .eq("account_id", user.id)
   189→        .limit(500);
   190→
   191→      if (!orders || orders.length === 0) { setTopRecommended([]); setRecLoading(false); return; }
   192→
   193→      const productCounts = {};
   194→      for (const order of orders) {
   195→        if (order.items && Array.isArray(order.items)) {
   196→          for (const item of order.items) {
   197→            if (item.product_id) {
   198→              productCounts[item.product_id] = (productCounts[item.product_id] || 0) + 1;
   199→            }
   200→          }
   201→        }
   202→      }
   203→
   204→      const topIds = Object.entries(productCounts)
   205→        .sort((a, b) => b[1] - a[1])
   206→        .slice(0, 5)
   207→        .map(([id, count]) => ({ id, count }));
   208→
   209→      if (topIds.length === 0) { setTopRecommended([]); setRecLoading(false); return; }
   210→
   211→      const { data: prods } = await supabase
   212→        .from("products")
   213→        .select("id, name, price, category, image_urls, stock")
   214→        .in("id", topIds.map(t => t.id));
   215→
   216→      const merged = topIds.map(t => {
   217→        const prod = prods?.find(p => p.id === t.id);
   218→        return prod ? { ...prod, rec_count: t.count } : null;
   219→      }).filter(Boolean);
   220→
   221→      setTopRecommended(merged);
   222→    } catch (err) {
   223→      console.error("Failed to fetch top recommended:", err);
   224→    }
   225→    setRecLoading(false);
   226→  }, [supabase]);
   227→
   228→  useEffect(() => { fetchTopRecommended(); }, [fetchTopRecommended]);

   231→  const addVariant = () => {

   247→    });
   248→  };

   265→    reader.readAsDataURL(file);
   266→  };
   267→
   268→  const handleDragOver = (e) => { e.preventDefault(); setDragOver(true); };
   269→  const handleDragLeave = (e) => { e.preventDefault(); setDragOver(false); };
   270→  const handleDrop = (e) => {
   271→    e.preventDefault();
   272→    setDragOver(false);
   273→    handleFileSelect(e.dataTransfer.files[0]);
   274→  };
   275→
   276→  const handleGenerateImage = async () => {
   277→    const form = document.querySelector('.modal form');
   278→    const productName = form?.elements?.name?.value;
   279→    const description = form?.elements?.description?.value;
   280→
   281→    if (!productName?.trim()) {
   282→      toast.warning("Please enter a product name first, then generate an image.");
   283→      return;
   284→    }
   285→
   286→    setGeneratingImage(true);
   287→    try {
   288→      const res = await fetch("/api/products/generate-image", {
   289→        method: "POST",
   290→        headers: { "Content-Type": "application/json" },
   291→        body: JSON.stringify({
   292→          product_name: productName.trim(),
   293→          description: description?.trim() || "",
   294→          style: aiStyle,
   295→        }),

   298→      const data = await res.json();

   300→        toast.error(data.error || "Image generation failed. Please try again.");
   301→        return;
   302→      }
   303→
   304→      const dataUrl = `data:image/png;base64,${data.image_base64}`;
   305→      setImagePreview(dataUrl);
   306→      setImageFile(null);
   307→      setGeneratedImageUrl(data.image_url);
   308→    } catch (err) {
   309→      console.error("Image generation error:", err);
   310→      toast.error("Image generation failed. Please try again.");
   311→    } finally {
   312→      setGeneratingImage(false);
   313→    }
   314→  };
   315→
   316→  const handleAddProduct = async (e) => {
   317→    e.preventDefault();
   318→    setSaving(true);
   319→    const fd = new FormData(e.target);
   320→    const { data: { user } } = await supabase.auth.getUser();
   321→
   322→    let imageUrl = null;
   323→
   324→    if (generatedImageUrl) {
   325→      imageUrl = generatedImageUrl;
   326→    } else if (imageFile) {
   327→      try {
   328→        await fetch("/api/storage/ensure-buckets", { method: "POST" });
   329→      } catch (e) {}
   330→
   331→      const ext = imageFile.name.split(".").pop();
   332→      const fileName = `${user.id}/${Date.now()}.${ext}`;
   333→
   334→      let uploadError;
   335→      let uploadData;
   336→
   337→      const clientResult = await supabase.storage
   338→        .from("product-images")
   339→        .upload(fileName, imageFile, { cacheControl: "3600", upsert: false });
   340→      uploadError = clientResult.error;
   341→      uploadData = clientResult.data;
   342→
   343→      if (uploadError) {
   344→        console.warn("[Products] Client upload failed, trying admin:", uploadError.message);
   345→        try {
   346→          const formData = new FormData();
   347→          formData.append("file", imageFile);
   348→          formData.append("path", fileName);
   349→          formData.append("bucket", "product-images");
   350→          // Pass auth token for server-side verification
   351→          const { data: sessData } = await supabase.auth.getSession();
   352→          const authToken = sessData.session?.access_token || '';
   353→          const adminRes = await fetch("/api/storage/upload", {
   354→            method: "POST",
   355→            headers: { 'Authorization': `Bearer ${authToken}` },
   356→            body: formData,
   357→          });
   358→          if (adminRes.ok) {
   359→            const adminData = await adminRes.json();
   360→            if (adminData.url) {
   361→              imageUrl = adminData.url;
   362→              uploadError = null;
   363→            }
   364→          }
   365→        } catch (adminErr) {
   366→          console.error("[Products] Admin upload also failed:", adminErr.message);
   367→        }
   368→      }
   369→
   370→      if (!uploadError && uploadData) {
   371→        const { data: urlData } = supabase.storage
   372→          .from("product-images")
   373→          .getPublicUrl(fileName);
   374→        imageUrl = urlData.publicUrl;
   375→      } else if (uploadError && !imageUrl) {
   376→        toast.error("Image upload failed: " + (uploadError.message || "Unknown error."));
   377→      }
   378→    }
   379→
   380→    // Build variants array: clean up empty entries
   381→    const hasVariants = variants.some((v) => v.name.trim() !== "");
   382→    const cleanVariants = variants
   383→      .filter((v) => v.name.trim() !== "")
   384→      .map((v) => ({
   385→        name: v.name.trim(),
   386→        sku: v.sku.trim() || null,
   387→        price: Number(v.price) || 0,
   388→        stock: Number(v.stock) || 0,
   389→        image_url: v.image_url.trim() || null,
   390→      }));
   391→
   392→    // Calculate base price/stock from variants or form fields
   393→    let basePrice, baseStock;
   394→    if (hasVariants) {
   395→      // When variants exist, use the lowest variant price as the base price
   396→      // and sum all variant stocks as the total stock
   397→      basePrice = cleanVariants.length > 0 ? Math.min(...cleanVariants.map(v => v.price)) : 0;
   398→      baseStock = cleanVariants.reduce((sum, v) => sum + v.stock, 0);
   399→    } else {
   400→      basePrice = parseFloat(fd.get("price"));
   401→      baseStock = parseInt(fd.get("stock"));
   402→    }
   403→
   404→    const payload = {
   405→      account_id: user.id,
   406→      name: fd.get("name"),
   407→      description: fd.get("description"),
   408→      price: basePrice,
   409→      category: fd.get("category"),
   410→      stock: baseStock,
   411→      status: fd.get("status") || "active",
   412→      variants: cleanVariants,
   413→    };
   414→
   415→    if (imageUrl) {
   416→      payload.image_urls = [imageUrl];
   417→    } else if (editingProduct && !imagePreview) {
   418→      payload.image_urls = [];
   419→    }
   420→
   421→    let error;
   422→    if (editingProduct) {
   423→      const { error: updateError } = await supabase.from("products").update(payload).eq("id", editingProduct.id).eq("account_id", user.id);
   424→      error = updateError;
   425→    } else {
   426→      if (!imageUrl && !editingProduct) {
   427→        payload.image_urls = [];
   428→      }
   429→      const { error: insertError } = await supabase.from("products").insert(payload);
   430→      error = insertError;
   431→    }
   432→
   433→    if (!error) {
   434→      closeModal();
   435→      fetchProducts();
   436→    }
   437→    setSaving(false);
   438→  };
   439→
   440→  const handleEdit = async (product) => {
   441→    setEditingProduct(product);
   442→    setImagePreview(product.image_urls && product.image_urls.length > 0 ? product.image_urls[0] : null);
   443→    // Load existing variants
   444→    const existingVariants = Array.isArray(product.variants) ? product.variants : [];
   445→    setVariants(
   446→      existingVariants.map((v) => ({
   447→        name: v.name || "",
   448→        sku: v.sku || "",
   449→        price: v.price ?? v.price_offset ?? 0,
   450→        stock: v.stock ?? 0,
   451→        image_url: v.image_url || "",
   452→      }))
   453→    );
   454→    setShowModal(true);
   455→  };
   456→
   457→  const handleDelete = async (id, imageUrls) => {
   458→    if (!(await confirmAction("Delete this product?"))) return;
   459→
   460→    if (imageUrls && imageUrls.length > 0) {
   461→      try {
   462→        const { data: { user } } = await supabase.auth.getUser();
   463→        if (user) {
   464→          const pathsToRemove = imageUrls.map(url => {
   465→            const parts = url.split("product-images/");
   466→            return parts.length > 1 ? parts[1] : null;
   467→          }).filter(Boolean);
   468→
   469→          if (pathsToRemove.length > 0) {
   470→            await supabase.storage.from("product-images").remove(pathsToRemove);
   471→          }
   472→        }
   473→      } catch (err) {
   474→        console.error("Failed to delete product images from storage:", err);
   475→      }
   476→    }
   477→
   478→    const { data: { user: delUser } } = await supabase.auth.getUser();
   479→    await supabase.from("products").delete().eq("id", id).eq("account_id", delUser?.id);
   480→    fetchProducts();
   481→  };
   482→
   483→  const closeModal = () => {
   484→    setShowModal(false);
   485→    setEditingProduct(null);
   486→    setImageFile(null);
   487→    setImagePreview(null);
   488→    setGeneratingImage(false);
   489→    setAiStyle("studio");
   490→    setGeneratedImageUrl(null);
   491→    setGeneratingDesc(false);
   492→    setAiDescEnglish("");
   493→    setAiDescArabic("");
   494→    setAiPriceSuggestion("");
   495→    setVariants([]);
   496→  };
   497→
   498→  const handleGenerateDescription = async () => {
   499→    const form = document.querySelector('.modal form');
   500→    const productName = form?.elements?.name?.value;
   501→    const category = form?.elements?.category?.value;
   502→
   503→    if (!productName?.trim()) {
   504→      toast.warning("Please enter a product name first, then generate a description.");
   505→      return;
   506→    }
   507→
   508→    setGeneratingDesc(true);
   509→    try {
   510→      const res = await fetch("/api/ai/generate-description", {
   511→        method: "POST",
   512→        headers: { "Content-Type": "application/json" },
   513→        body: JSON.stringify({
   514→          product_name: productName.trim(),
   515→          features: productName.trim(),
   516→          category: category || "General",
   517→        }),
   518→      });
   519→
   520→      const data = await res.json();
   521→      if (!res.ok || !data.success) {
   522→        toast.error(data.error || "Description generation failed.");
   523→        return;
   524→      }
   525→
   526→      if (data.english) {
   527→        setAiDescEnglish(data.english);
   528→        const descField = form?.elements?.description;
   529→        if (descField) descField.value = data.english;
   530→      }
   531→      if (data.arabic) setAiDescArabic(data.arabic);
   532→      if (data.price_suggestion) setAiPriceSuggestion(data.price_suggestion);
   533→    } catch (err) {
   534→      console.error("Description generation error:", err);
   535→      toast.error("Description generation failed. Please try again.");
   536→    } finally {
   537→      setGeneratingDesc(false);
   538→    }
   539→  };
   540→
   541→  const emojis = { Bags: "👜", Jewelry: "💎", Accessories: "🧣", Electronics: "📱", Watches: "⌚", Clothing: "👗" };
   542→
   543→  return (
   544→    <>
   545→      <div className="page-header">
   546→        <h1>Products</h1>
   547→        <div className="page-header-actions">
   548→          {limitReached ? (
   549→            <button className="btn btn-primary" onClick={() => router.push('/dashboard/billing')} id="add-product" style={{ opacity: 0.7 }}>
   550→               Upgrade to Add More
   551→            </button>
   552→          ) : (
   553→            <button className="btn btn-primary" onClick={() => { setVariants([]); setShowModal(true); }} id="add-product">
   554→              <Plus size={16} /> Add Product
   555→            </button>
   556→          )}
   557→        </div>
   558→      </div>
   559→
   560→      <div className="filter-bar">
   561→        <div className="filter-tabs">
   562→          {[
   563→            { key: "all", label: "All Products" },
   564→            { key: "active", label: "Active" },
   565→            { key: "draft", label: "Draft" },
   566→            { key: "low", label: "Low Stock" },
   567→          ].map((f) => (
   568→            <button key={f.key} className={`filter-tab ${filter === f.key ? "active" : ""}`} onClick={() => setFilter(f.key)}>
   569→              {f.label}
   570→            </button>
   571→          ))}
   572→        </div>
