    94→
    95→export default function SegmentsPage() {
    96→  const toast = useToast();
    97→  const router = useRouter();
    98→  
    99→  const confirmAction = useConfirm();
   100→  const [segments, setSegments] = useState([]);
   101→  const [loading, setLoading] = useState(true);
   102→  const [showCreateModal, setShowCreateModal] = useState(false);
   103→  const [saving, setSaving] = useState(false);
   104→  const [selectedSegment, setSelectedSegment] = useState(null);
   105→  const [segmentCustomers, setSegmentCustomers] = useState([]);
   106→  const [customersLoading, setCustomersLoading] = useState(false);
   107→  const [computing, setComputing] = useState(false);
   108→  const [previewCount, setPreviewCount] = useState(null);
   109→  const [previewLoading, setPreviewLoading] = useState(false);
   110→
   111→  const [form, setForm] = useState({
   112→    name: "",
   113→    description: "",
   114→    color: "#5865F2",
   115→    icon: "Users",
   116→    rules: { operator: "AND", conditions: [] },
   117→  });
   118→
   119→  const supabase = createClient();
   120→
   121→  const fetchSegments = useCallback(async () => {
   122→    setLoading(true);
   123→    try {
   124→      const res = await fetch("/api/segments");
   125→      const data = await res.json();
   126→      if (data.success) {
   127→        setSegments(data.segments || []);
   128→      }
   129→    } catch (err) {
   130→      console.error("Failed to fetch segments:", err);
   131→    }
   132→    setLoading(false);
   133→  }, []);
   134→
   135→  useEffect(() => {
   136→    let cancelled = false;
   137→    const load = async () => {
   138→      setLoading(true);
   139→      try {
   140→        const res = await fetch("/api/segments");
   141→        const data = await res.json();
   142→        if (!cancelled && data.success) {
   143→          setSegments(data.segments || []);
   144→        }
   145→      } catch (err) {
   146→        console.error("Failed to fetch segments:", err);
   147→      }
   148→      if (!cancelled) setLoading(false);
   149→    };
   150→    load();
   151→    return () => { cancelled = true; };
   152→  }, []);
   153→
   154→  // Preview count when rules change
   155→  const fetchPreviewCount = useCallback(async () => {
   156→    if (!form.rules.conditions || form.rules.conditions.length === 0) {
   157→      setPreviewCount(null);
   158→      return;
   159→    }
   160→    setPreviewLoading(true);
   161→    try {
   162→      const { data: { user } } = await supabase.auth.getUser();
   163→      if (!user) return;
   164→
   165→      let query = supabase.from("customers").select("id", { count: "exact", head: true }).eq("account_id", user.id);
   166→      for (const cond of form.rules.conditions) {
   167→        if (!cond.field || !cond.operator || cond.value === undefined || cond.value === "") continue;
   168→        switch (cond.operator) {
   169→          case "greater_than": query = query.gte(cond.field, Number(cond.value)); break;
   170→          case "less_than": query = query.lte(cond.field, Number(cond.value)); break;
   171→          case "equals": query = cond.field === "tags" ? query.contains("tags", [cond.value]) : query.eq(cond.field, cond.value); break;
   172→          case "contains": query = cond.field === "tags" ? query.contains("tags", [cond.value]) : query.ilike(cond.field, `%${cond.value}%`); break;
   173→          case "within_days": { const d = new Date(); d.setDate(d.getDate() - Number(cond.value)); query = query.gte(cond.field, d.toISOString()); break; }
   174→          case "not_within_days": { const d = new Date(); d.setDate(d.getDate() - Number(cond.value)); query = query.lt(cond.field, d.toISOString()); break; }
   175→        }
   176→      }
   177→      const { count } = await query;
   178→      setPreviewCount(count || 0);
   179→    } catch (err) {
   180→      console.error("Preview count error:", err);
   181→    }
   182→    setPreviewLoading(false);
   183→  }, [form.rules, supabase]);
   184→
   185→  useEffect(() => {
   186→    if (showCreateModal) {
   187→      const debounce = setTimeout(fetchPreviewCount, 400);
   188→      return () => clearTimeout(debounce);
   189→    }
   190→  }, [showCreateModal, fetchPreviewCount]);
   191→
   192→  const handleCreateSegment = async (e) => {
   193→    e.preventDefault();
   194→    if (!form.name.trim()) return;
   195→    setSaving(true);
   196→    try {
   197→      const res = await fetch("/api/segments", {
   198→        method: "POST",
   199→        headers: { "Content-Type": "application/json" },
   200→        body: JSON.stringify(form),
   201→      });
   202→      const data = await res.json();
   203→      if (data.success) {
   204→        setShowCreateModal(false);
   205→        resetForm();
   206→        fetchSegments();
   207→      } else {
   208→        toast.error(data.error || "Failed to create segment");
   209→      }
   210→    } catch (err) {
   211→      toast.error("Failed to create segment");
   212→    }
   213→    setSaving(false);
   214→  };
   215→
   216→  const handleCreateFromTemplate = async (template) => {
   217→    setSaving(true);
   218→    try {
   219→      const res = await fetch("/api/segments", {
   220→        method: "POST",
   221→        headers: { "Content-Type": "application/json" },
   222→        body: JSON.stringify({
   223→          name: template.name,
   224→          description: template.description,
   225→          color: template.color,
   226→          icon: template.icon,
   227→          rules: template.rules,
   228→        }),
   229→      });
   230→      const data = await res.json();
   231→      if (data.success) {
   232→        fetchSegments();
   233→      } else {
   234→        toast.error(data.error || "Failed to create segment");
   235→      }
   236→    } catch (err) {
   237→      toast.error("Failed to create segment from template");
   238→    }
   239→    setSaving(false);
   240→  };
   241→
   242→  const handleDelete = async (id) => {
   243→    if (!(await confirmAction("Delete this segment?"))) return;
   244→    try {
   245→      const res = await fetch(`/api/segments/${id}`, { method: "DELETE" });
   246→      if (res.ok) {
   247→        if (selectedSegment?.id === id) setSelectedSegment(null);
   248→        fetchSegments();
   249→      }
   250→    } catch (err) {
   251→      console.error("Delete error:", err);
   252→    }
   253→  };
   254→
   255→  const handleCompute = async (segment) => {
   256→    setComputing(true);
   257→    try {
   258→      const res = await fetch(`/api/segments/${segment.id}/compute`, { method: "POST" });
   259→      const data = await res.json();
   260→      if (data.success) {
   261→        fetchSegments();
   262→        if (selectedSegment?.id === segment.id) {
   263→          fetchSegmentDetail(segment.id);
   264→        }
   265→      }
   266→    } catch (err) {
   267→      console.error("Compute error:", err);
   268→    }
   269→    setComputing(false);
   270→  };
   271→
   272→  const fetchSegmentDetail = async (id) => {
   273→    setCustomersLoading(true);
   274→    try {
   275→      const res = await fetch(`/api/segments/${id}/customers?limit=20`);
   276→      const data = await res.json();
   277→      if (data.success) {
   278→        setSegmentCustomers(data.customers || []);
   279→      }
   280→    } catch (err) {
   281→      console.error("Fetch segment detail error:", err);
   282→    }
   283→    setCustomersLoading(false);
   284→  };
   285→
   286→  const handleSelectSegment = (segment) => {
   287→    setSelectedSegment(segment);
   288→    fetchSegmentDetail(segment.id);
   289→  };
   290→
   291→  const addCondition = () => {
   292→    setForm({
   293→      ...form,
   294→      rules: {
   295→        ...form.rules,
   296→        conditions: [...(form.rules.conditions || []), { field: "total_spent", operator: "greater_than", value: "" }],
   297→      },
   298→    });
   299→  };
   300→
   301→  const removeCondition = (index) => {
   302→    const newConditions = form.rules.conditions.filter((_, i) => i !== index);
   303→    setForm({ ...form, rules: { ...form.rules, conditions: newConditions } });
   304→  };
   305→
   306→  const updateCondition = (index, key, value) => {
   307→    const newConditions = [...form.rules.conditions];
   308→    newConditions[index] = { ...newConditions[index], [key]: value };
   309→    // Reset operator when field type changes
   310→    if (key === "field") {
   311→      const fieldType = FIELD_OPTIONS.find(f => f.value === value)?.type || "number";
   312→      const ops = OPERATOR_OPTIONS[fieldType] || [];
   313→      newConditions[index].operator = ops[0]?.value || "greater_than";
   314→      newConditions[index].value = "";
   315→    }
   316→    setForm({ ...form, rules: { ...form.rules, conditions: newConditions } });
   317→  };
   318→
   319→  const resetForm = () => {
   320→    setForm({ name: "", description: "", color: "#5865F2", icon: "Users", rules: { operator: "AND", conditions: [] } });
   321→    setPreviewCount(null);
   322→  };
   323→
   324→  const formatRelative = (d) => {
   325→    if (!d) return "Never";
   326→    const diff = new Date().getTime() - new Date(d).getTime();
   327→    const mins = Math.floor(diff / 60000);
   328→    if (mins < 1) return "Just now";
   329→    if (mins < 60) return `${mins}m ago`;
   330→    const hours = Math.floor(mins / 60);
   331→    if (hours < 24) return `${hours}h ago`;
   332→    return `${Math.floor(hours / 24)}d ago`;
   333→  };
   334→
   335→  return (
   336→    <>
   337→      <div className="page-header">
   338→        <h1>Segments</h1>
   339→        <div className="page-header-actions">
   340→          <button className="btn btn-primary" onClick={() => { resetForm(); setShowCreateModal(true); }}>
   341→            <Plus size={16} /> Create Segment
   342→          </button>
   343→        </div>
   344→      </div>
   345→
   346→      {/* Templates */}
   347→      <div style={{ marginBottom: "var(--space-xl)" }}>
   348→        <h3 style={{ fontSize: "var(--font-size-sm)", fontWeight: 600, color: "var(--text-tertiary)", textTransform: "uppercase", letterSpacing: 1, marginBottom: "var(--space-md)" }}>
   349→          Quick Templates
   350→        </h3>
   351→        <div style={{ display: "flex", gap: "var(--space-md)", flexWrap: "wrap" }}>
   352→          {SEGMENT_TEMPLATES.map((tmpl, i) => {
   353→            const IconComp = ICON_MAP[tmpl.icon] || Users;
   354→            return (
   355→              <button
   356→                key={i}
   357→                onClick={() => handleCreateFromTemplate(tmpl)}
   358→                disabled={saving}
   359→                style={{
   360→                  display: "flex", alignItems: "center", gap: 8, padding: "8px 16px",
   361→                  background: `${tmpl.color}12`, border: `1px solid ${tmpl.color}33`,
   362→                  borderRadius: "var(--radius-full)", cursor: "pointer", color: tmpl.color,
   363→                  fontSize: "var(--font-size-sm)", fontWeight: 600, fontFamily: "var(--font-family)",
   364→                  transition: "all 0.2s",
   365→                }}
   366→              >
   367→                <IconComp size={14} />
   368→                {tmpl.name}
   369→              </button>
   370→            );
   371→          })}
   372→        </div>
   373→      </div>
   374→
   375→      {/* Segments Grid */}
   376→      {loading ? (
   377→        <div style={{ padding: "var(--space-3xl)", textAlign: "center", color: "var(--text-tertiary)" }}>Loading segments...</div>
   378→      ) : (
   379→        <div style={{ display: "grid", gridTemplateColumns: selectedSegment ? "1fr 1fr" : "repeat(auto-fill, minmax(300px, 1fr))", gap: "var(--space-lg)" }}>
   380→          {segments.map((segment) => {
   381→            const IconComp = ICON_MAP[segment.icon] || Users;
   382→            const isSelected = selectedSegment?.id === segment.id;
   383→            return (
   384→              <div
   385→                key={segment.id}
   386→                className="dashboard-panel"
   387→                style={{
   388→                  cursor: "pointer",
   389→                  borderColor: isSelected ? segment.color : undefined,
   390→                  boxShadow: isSelected ? `0 0 20px ${segment.color}22` : undefined,
   391→                }}
   392→                onClick={() => handleSelectSegment(segment)}
   393→              >
   394→                <div style={{ padding: "var(--space-xl)" }}>
   395→                  <div style={{ display: "flex", alignItems: "flex-start", justifyContent: "space-between", marginBottom: "var(--space-md)" }}>
   396→                    <div style={{ display: "flex", alignItems: "center", gap: "var(--space-md)" }}>
   397→                      <div style={{
   398→                        width: 44, height: 44, borderRadius: "var(--radius-md)",
   399→                        background: `${segment.color}18`, color: segment.color,
   400→                        display: "flex", alignItems: "center", justifyContent: "center", flexShrink: 0,
   401→                      }}>
   402→                        <IconComp size={22} />
   403→                      </div>
   404→                      <div>
   405→                        <h3 style={{ fontWeight: 700, fontSize: "var(--font-size-base)" }}>{segment.name}</h3>
   406→                        {segment.description && (
   407→                          <p style={{ fontSize: "var(--font-size-xs)", color: "var(--text-tertiary)", marginTop: 2 }}>{segment.description}</p>
   408→                        )}
   409→                      </div>
   410→                    </div>
   411→                    <div style={{ display: "flex", gap: 4 }}>
   412→                      <button
   413→                        className="topbar-btn"
   414→                        title="Recompute"
   415→                        onClick={(e) => { e.stopPropagation(); handleCompute(segment); }}
   416→                        style={{ width: 28, height: 28 }}
   417→                      >
   418→                        <RefreshCw size={13} className={computing ? "spin" : ""} />
   419→                      </button>
   420→                      <button
   421→                        className="topbar-btn"
   422→                        title="Delete"
   423→                        onClick={(e) => { e.stopPropagation(); handleDelete(segment.id); }}
   424→                        style={{ width: 28, height: 28, color: "var(--accent-red)" }}
   425→                      >
   426→                        <Trash2 size={13} />
   427→                      </button>
   428→                    </div>
   429→                  </div>
   430→
   431→                  <div style={{ display: "flex", alignItems: "center", gap: "var(--space-lg)", marginTop: "var(--space-lg)" }}>
   432→                    <div>
   433→                      <div style={{ fontSize: "var(--font-size-3xl)", fontWeight: 800, color: segment.color, lineHeight: 1.2 }}>
   434→                        {segment.customer_count || 0}
   435→                      </div>
   436→                      <div style={{ fontSize: "var(--font-size-xs)", color: "var(--text-tertiary)", marginTop: 2 }}>customers</div>
   437→                    </div>
   438→                    <div style={{ display: "flex", flexDirection: "column", gap: 4, marginLeft: "auto" }}>
   439→                      <span style={{
   440→                        padding: "2px 10px", borderRadius: 20, fontSize: 10, fontWeight: 600,
   441→                        background: segment.is_dynamic ? "rgba(0,210,255,0.1)" : "rgba(255,255,255,0.05)",
   442→                        color: segment.is_dynamic ? "var(--accent-secondary)" : "var(--text-tertiary)",
   443→                        border: `1px solid ${segment.is_dynamic ? "rgba(0,210,255,0.2)" : "var(--border-subtle)"}`,
   444→                        textAlign: "center",
   445→                      }}>
   446→                        {segment.is_dynamic ? "Dynamic" : "Static"}
   447→                      </span>
   448→                      <span style={{ fontSize: 10, color: "var(--text-tertiary)", textAlign: "center" }}>
   449→                        {formatRelative(segment.last_computed_at)}
   450→                      </span>
   451→                    </div>
   452→                  </div>
   453→
   454→                  {/* Rules preview */}
   455→                  {segment.rules?.conditions?.length > 0 && (
   456→                    <div style={{ marginTop: "var(--space-md)", padding: "var(--space-sm) var(--space-md)", background: "var(--bg-glass)", borderRadius: "var(--radius-sm)", border: "1px solid var(--border-subtle)" }}>
   457→                      <div style={{ display: "flex", alignItems: "center", gap: 4, marginBottom: 4 }}>
   458→                        <Filter size={11} style={{ color: "var(--text-tertiary)" }} />
   459→                        <span style={{ fontSize: 10, color: "var(--text-tertiary)", fontWeight: 600, textTransform: "uppercase" }}>
   460→                          {segment.rules.operator} Rules
   461→                        </span>
   462→                      </div>
   463→                      <div style={{ display: "flex", gap: 4, flexWrap: "wrap" }}>
   464→                        {segment.rules.conditions.map((cond, ci) => (
   465→                          <span key={ci} style={{
   466→                            padding: "1px 6px", borderRadius: 4, fontSize: 10, fontWeight: 500,
   467→                            background: `${segment.color}10`, color: segment.color,
   468→                          }}>
   469→                            {FIELD_OPTIONS.find(f => f.value === cond.field)?.label || cond.field} {OPERATOR_OPTIONS[FIELD_OPTIONS.find(f => f.value === cond.field)?.type || "number"]?.find(o => o.value === cond.operator)?.label || cond.operator} {cond.value}
   470→                          </span>
   471→                        ))}
   472→                      </div>
   473→                    </div>
   474→                  )}
   475→                </div>
   476→              </div>
   477→            );
   478→          })}
   479→
   480→          {/* Segment Detail Panel */}
   481→          {selectedSegment && (
   482→            <div className="dashboard-panel" style={{ gridColumn: "span 1" }}>
   483→              <div style={{ padding: "var(--space-xl)" }}>
   484→                <div style={{ display: "flex", alignItems: "center", justifyContent: "space-between", marginBottom: "var(--space-lg)" }}>
   485→                  <div style={{ display: "flex", alignItems: "center", gap: "var(--space-sm)" }}>
   486→                    <button className="topbar-btn" style={{ width: 28, height: 28 }} onClick={() => setSelectedSegment(null)}>
   487→                      <ChevronLeft size={16} />
   488→                    </button>
   489→                    <h3 style={{ fontWeight: 700 }}>Customers in {selectedSegment.name}</h3>
   490→                  </div>
   491→                  <div style={{ display: "flex", gap: "var(--space-sm)" }}>
   492→                    <button
   493→                      className="btn btn-secondary btn-sm"
   494→                      onClick={() => handleCompute(selectedSegment)}
   495→                      disabled={computing}
   496→                    >
   497→                      <RefreshCw size={12} className={computing ? "spin" : ""} /> Recompute
   498→                    </button>
   499→                    <button
   500→                      className="btn btn-primary btn-sm"
   501→                      onClick={() => router.push(`/dashboard/campaigns?segment=${selectedSegment.id}`)}
   502→                    >
   503→                      <Megaphone size={12} /> Send Campaign
   504→                    </button>
   505→                  </div>
   506→                </div>
   507→
   508→                {customersLoading ? (
   509→                  <div style={{ textAlign: "center", padding: "var(--space-xl)", color: "var(--text-tertiary)" }}>
   510→                    <Loader2 size={16} className="spin" style={{ display: "inline-block" }} /> Loading...
   511→                  </div>
   512→                ) : segmentCustomers.length === 0 ? (
   513→                  <div style={{ textAlign: "center", padding: "var(--space-xl)", color: "var(--text-tertiary)", fontSize: "var(--font-size-sm)" }}>

   531→                              <div style={{ display: "flex", alignItems: "center", gap: 8 }}>

   533→                                  width: 28, height: 28, borderRadius: "50%", background: "var(--accent-gradient)",
   534→                                  display: "flex", alignItems: "center", justifyContent: "center",
   535→                                  fontSize: 10, fontWeight: 700, flexShrink: 0,

   537→                                  {c.name?.split(" ").map(n => n[0]).join("") || "?"}

   540→                                  <div style={{ fontWeight: 600, fontSize: "var(--font-size-sm)" }}>{c.name || "Unknown"}</div>
   541→                                  <div style={{ fontSize: 10, color: "var(--text-tertiary)" }}>{c.email || c.phone || ""}</div>
   542→                                </div>
   543→                              </div>
   544→                            </td>

   546→                              <span className={`channel-badge ${c.channel || "whatsapp"}`} style={{ fontSize: 10 }}>
   547→                                {c.channel || "N/A"}
   548→                              </span>
   549→                            </td>
   550→                            <td style={{ fontWeight: 600, color: "var(--accent-primary-light)" }}>{(c.total_spent || 0).toLocaleString()} EGP</td>
   551→                            <td>{c.total_orders || 0}</td>
   552→                          </tr>
   553→                        ))}
   554→                      </tbody>
   555→                    </table>
   556→                  </div>
   557→                )}
   558→              </div>
   559→            </div>
   560→          )}
   561→
   562→          {segments.length === 0 && !loading && (
   563→            <div className="empty-state" style={{ gridColumn: "1 / -1" }}>
   564→              <div className="empty-state-icon"><Target size={32} /></div>

   568→                <Plus size={16} /> Create Segment
   569→              </button>
   570→            </div>
   571→          )}
   572→        </div>
   573→      )}
   574→
   575→      {/* Create Segment Modal */}
   576→      {showCreateModal && (
   577→        <div className="modal-overlay" onClick={(e) => e.target === e.currentTarget && setShowCreateModal(false)}>
   578→          <div className="modal" style={{ maxWidth: 640 }}>
   579→            <div className="modal-header">
   580→              <h3>Create Segment</h3>
   581→              <button className="modal-close" onClick={() => setShowCreateModal(false)}><X size={18} /></button>

   583→            <form onSubmit={handleCreateSegment}>
   584→              <div className="modal-body">
   585→                <div className="form-group">
   586→                  <label className="form-label">Segment Name</label>
   587→                  <input
   588→                    type="text" className="form-input" placeholder="e.g. VIP Customers"
   589→                    value={form.name} onChange={(e) => setForm({ ...form, name: e.target.value })} required

   591→                </div>
   592→
   593→                <div className="form-group">
   594→                  <label className="form-label">Description</label>
   595→                  <textarea
   596→                    className="form-input form-textarea" placeholder="Describe this segment..."
   597→                    value={form.description} onChange={(e) => setForm({ ...form, description: e.target.value })}
   598→                    rows={2}
   599→                  />
   600→                </div>
   601→
   602→                <div style={{ display: "grid", gridTemplateColumns: "1fr 1fr", gap: "var(--space-md)" }}>

   604→                  <div className="form-group">

   609→                          key={c.value} type="button"
   610→                          onClick={() => setForm({ ...form, color: c.value })}

   612→                            width: 28, height: 28, borderRadius: "50%", background: c.value,
   613→                            border: form.color === c.value ? "3px solid white" : "2px solid transparent",
   614→                            cursor: "pointer", outline: "none", transition: "all 0.15s",
   615→                            boxShadow: form.color === c.value ? `0 0 12px ${c.value}66` : "none",
   616→                          }}
   617→                          title={c.name}
   618→                        />
   619→                      ))}
   620→                    </div>
   621→                  </div>
   622→
   623→                  {/* Icon Picker */}
   624→                  <div className="form-group">
   625→                    <label className="form-label">Icon</label>
   626→                    <div style={{ display: "flex", gap: 6, flexWrap: "wrap" }}>
   627→                      {Object.entries(ICON_MAP).map(([name, Comp]) => (
   628→                        <button
   629→                          key={name} type="button"
   630→                          onClick={() => setForm({ ...form, icon: name })}
   631→                          style={{
   632→                            width: 32, height: 32, borderRadius: "var(--radius-sm)",
   633→                            background: form.icon === name ? `${form.color}22` : "var(--bg-glass)",
   634→                            border: form.icon === name ? `1px solid ${form.color}44` : "1px solid var(--border-subtle)",
   635→                            color: form.icon === name ? form.color : "var(--text-tertiary)",
   636→                            cursor: "pointer", display: "flex", alignItems: "center", justifyContent: "center",
   637→                            transition: "all 0.15s",
   638→                          }}
   639→                        >
   640→                          <Comp size={16} />
   641→                        </button>
   642→                      ))}
   643→                    </div>
   644→                  </div>
   645→                </div>
   646→
   647→                {/* Rule Builder */}
   648→                <div className="form-group">
   649→                  <label className="form-label" style={{ display: "flex", alignItems: "center", justifyContent: "space-between" }}>
   650→                    <span>Segmentation Rules</span>

   652→                      <Plus size={12} /> Add Condition
   653→                    </button>
   654→                  </label>
   655→
   656→                  {/* AND/OR toggle */}
   657→                  {form.rules.conditions.length > 1 && (

   659→                      {["AND", "OR"].map((op) => (

   661→                          key={op} type="button"

   675→
   676→                  <div style={{ display: "flex", flexDirection: "column", gap: "var(--space-sm)" }}>
   677→                    {form.rules.conditions.map((cond, index) => {

   679→                      const operators = OPERATOR_OPTIONS[fieldType] || [];
   680→                      return (
   681→                        <div key={index} style={{
   682→                          display: "flex", gap: 6, alignItems: "center", padding: "var(--space-sm) var(--space-md)",
   683→                          background: "var(--bg-glass)", borderRadius: "var(--radius-md)", border: "1px solid var(--border-subtle)",

   685→                          <select
   686→                            className="form-input" style={{ flex: 1, padding: "6px 8px", fontSize: "var(--font-size-xs)" }}
   687→                            value={cond.field}
   688→                            onChange={(e) => updateCondition(index, "field", e.target.value)}

   705→                            onChange={(e) => updateCondition(index, "value", e.target.value)}

   708→                            type="button" onClick={() => removeCondition(index)}
   709→                            style={{ background: "none", border: "none", color: "var(--accent-red)", cursor: "pointer", padding: 4 }}
   710→                          >
   711→                            <X size={14} />
   712→                          </button>
   713→                        </div>
   714→                      );
   715→                    })}
   716→                    {form.rules.conditions.length === 0 && (
   717→                      <div style={{ padding: "var(--space-lg)", textAlign: "center", color: "var(--text-tertiary)", fontSize: "var(--font-size-sm)", border: "1px dashed var(--border-medium)", borderRadius: "var(--radius-md)" }}>
   718→                        No conditions added. All customers will be included.
   719→                      </div>
   720→                    )}
   721→                  </div>
   722→
   723→                  {/* Preview count */}
   724→                  {form.rules.conditions.length > 0 && (
   725→                    <div style={{
   726→                      marginTop: "var(--space-md)", padding: "var(--space-sm) var(--space-md)",
   727→                      background: `${form.color}12`, borderRadius: "var(--radius-sm)", border: `1px solid ${form.color}22`,
   728→                      fontSize: "var(--font-size-sm)", color: form.color, display: "flex", alignItems: "center", gap: 6,
   729→                    }}>
   730→                      <Users size={14} />
   731→                      {previewLoading ? "Counting..." : `${previewCount ?? "?"} customers match these rules`}
   732→                    </div>
   733→                  )}
   734→                </div>
   735→              </div>
   736→              <div className="modal-footer">
   737→                <button type="button" className="btn btn-secondary" onClick={() => setShowCreateModal(false)}>Cancel</button>
   738→                <button type="submit" className="btn btn-primary" disabled={saving || !form.name.trim()}>
   739→                  {saving ? <><Loader2 size={14} className="spin" /> Creating...</> : "Create Segment"}
   740→                </button>
   741→              </div>
   742→            </form>
   743→          </div>
   744→        </div>
   745→      )}
   746→    </>
   747→  );
