     1→"use client";
     2→
     3→import { useState, useEffect, useCallback } from "react";
     4→import { useRouter } from "next/navigation";
     5→import {
     6→  Plus, Send, Clock, Users, Eye, BarChart3, Pause, Trash2,
     7→  MessageCircle, X, Loader2, Calendar, Filter, Check, Zap,
     8→  Copy, AlertCircle, ChevronDown, ChevronUp, Globe, Camera, MessageSquare,
     9→} from "lucide-react";
    10→import { createClient } from "@/lib/supabase/client";
    11→import { useToast } from "../components/ToastProvider";
    12→import { useConfirm } from "../components/ConfirmProvider";
    13→import { getPlanLimits } from "@/lib/plan-limits";
    14→import { PageSkeleton } from "@/components/SkeletonLoader";
    15→
    16→const CHANNEL_ICON = {
    17→  instagram: <Camera size={12} />,
    18→  facebook: <Globe size={12} />,
    19→  whatsapp: <MessageSquare size={12} />,
    20→};
    21→
    22→export default function CampaignsPage() {
    23→  const toast = useToast();
    24→  
    25→  const confirmAction = useConfirm();
    26→  const [campaigns, setCampaigns] = useState([]);
    27→  const [filter, setFilter] = useState("all");
    28→  const [loading, setLoading] = useState(true);
    29→  const [accountPlan, setAccountPlan] = useState("starter");
    30→
    31→  // Create campaign modal
    32→  const [showCreateModal, setShowCreateModal] = useState(false);
    33→  const [saving, setSaving] = useState(false);
    34→  const [newCampaign, setNewCampaign] = useState({
    35→    name: "",
    36→    message: "",
    37→    audience: "all",
    38→    channel: "all",
    39→    tag: "all",
    40→    minSpent: "",
    41→    scheduledAt: "",
    42→    segmentId: "",
    43→  });
    44→
    45→  // Saved segments
    46→  const [savedSegments, setSavedSegments] = useState([]);
    47→
    48→  // Send campaign state
    49→  const [sendingCampaignId, setSendingCampaignId] = useState(null);
    50→  const [sendResult, setSendResult] = useState(null);
    51→
    52→  // Audience estimate
    53→  const [audienceCount, setAudienceCount] = useState(null);
    54→  const [estimating, setEstimating] = useState(false);
    55→
    56→  // Broadcast log detail modal
    57→  const [expandedCampaignId, setExpandedCampaignId] = useState(null);
    58→  const [broadcastLogs, setBroadcastLogs] = useState([]);
    59→  const [logSummary, setLogSummary] = useState(null);
    60→  const [loadingLogs, setLoadingLogs] = useState(false);
    61→
    62→  const supabase = createClient();
    63→  const router = useRouter();
    64→  const planLimits = getPlanLimits(accountPlan);
    65→
    66→  const fetchCampaigns = useCallback(async () => {
    67→    setLoading(true);
    68→    const { data: { user } } = await supabase.auth.getUser();
    69→    if (user) {
    70→      const { data: account } = await supabase.from("accounts").select("plan").eq("id", user.id).single();
    71→      if (account?.plan) setAccountPlan(account.plan);
    72→    }
    73→
    74→    let query = supabase.from("campaigns").select("*").order("created_at", { ascending: false });
    75→    if (filter !== "all") query = query.eq("status", filter);
    76→    const { data } = await query;
    77→    setCampaigns(data || []);
    78→    setLoading(false);
    79→  }, [filter]);
    80→
    81→  useEffect(() => { fetchCampaigns(); }, [fetchCampaigns]);
    82→
    83→  // Fetch saved segments when create modal opens
    84→  useEffect(() => {
    85→    if (!showCreateModal) return;
    86→    let cancelled = false;
    87→    fetch("/api/segments")
    88→      .then((res) => res.json())
    89→      .then((data) => {
    90→        if (!cancelled && data.success) setSavedSegments(data.segments || []);
    91→      })
    92→      .catch(() => {});
    93→    return () => { cancelled = true; };
    94→  }, [showCreateModal]);
    95→
    96→  // Estimate audience size when filters change
    97→  const estimateAudience = useCallback(async () => {
    98→    const { data: { user } } = await supabase.auth.getUser();
    99→    if (!user) return;
   100→
   101→    setEstimating(true);
   102→    let query = supabase.from("customers").select("id", { count: "exact", head: true }).eq("account_id", user.id);
   103→
   104→    if (newCampaign.channel && newCampaign.channel !== "all") {
   105→      query = query.eq("channel", newCampaign.channel);
   106→    }
   107→    if (newCampaign.tag && newCampaign.tag !== "all") {
   108→      query = query.contains("tags", [newCampaign.tag]);
   109→    }
   110→    if (newCampaign.minSpent && Number(newCampaign.minSpent) > 0) {
   111→      query = query.gte("total_spent", Number(newCampaign.minSpent));
   112→    }
   113→
   114→    const { count } = await query;
   115→    setAudienceCount(count || 0);
   116→    setEstimating(false);
   117→  }, [newCampaign.channel, newCampaign.tag, newCampaign.minSpent]);
   118→
   119→  useEffect(() => {
   120→    if (showCreateModal) {
   121→      estimateAudience();
   122→    }
   123→  }, [showCreateModal, estimateAudience]);
   124→
   125→  const handleDelete = async (id) => {
   126→    if (!(await confirmAction("Delete this campaign?"))) return;
   127→    await supabase.from("campaigns").delete().eq("id", id);
   128→    fetchCampaigns();
   129→  };
   130→
   131→  const handleStatusChange = async (id, newStatus) => {
   132→    await supabase.from("campaigns").update({ status: newStatus }).eq("id", id);
   133→    fetchCampaigns();
   134→  };
   135→
   136→  const handleCreateCampaign = async (e) => {
   137→    e.preventDefault();
   138→    if (!newCampaign.name.trim()) return;
   139→    setSaving(true);
   140→
   141→    const { data: { user } } = await supabase.auth.getUser();
   142→
   143→    // Build audience_filter JSONB
   144→    const audienceFilter = { type: newCampaign.audience };
   145→    if (newCampaign.channel && newCampaign.channel !== "all") {
   146→      audienceFilter.channel = newCampaign.channel;
   147→    }
   148→    if (newCampaign.tag && newCampaign.tag !== "all") {
   149→      audienceFilter.tag = newCampaign.tag;
   150→    }
   151→    if (newCampaign.minSpent && Number(newCampaign.minSpent) > 0) {
   152→      audienceFilter.min_spent = Number(newCampaign.minSpent);
   153→    }
   154→
   155→    let scheduledAt = null;
   156→    if (newCampaign.scheduledAt) {
   157→      scheduledAt = new Date(newCampaign.scheduledAt).toISOString();
   158→    }
   159→
   160→    // If a segment is selected, include it in the audience filter
   161→    if (newCampaign.segmentId) {
   162→      const selectedSeg = savedSegments.find(s => s.id === newCampaign.segmentId);
   163→      if (selectedSeg) {
   164→        audienceFilter.segment_id = newCampaign.segmentId;
   165→        audienceFilter.segment_name = selectedSeg.name;
   166→        // Merge segment rules into audience filter
   167→        if (selectedSeg.rules?.conditions?.length > 0) {
   168→          for (const cond of selectedSeg.rules.conditions) {
   169→            if (cond.field === "channel" && cond.operator === "equals") audienceFilter.channel = cond.value;
   170→            if (cond.field === "tags" && (cond.operator === "equals" || cond.operator === "contains")) audienceFilter.tag = cond.value;
   171→            if (cond.field === "total_spent" && cond.operator === "greater_than") audienceFilter.min_spent = Number(cond.value);
   172→          }
   173→        }
   174→      }
   175→    }
   176→
   177→    const { error } = await supabase.from("campaigns").insert({
   178→      account_id: user.id,
   179→      name: newCampaign.name.trim(),
   180→      message_template: newCampaign.message.trim(),
   181→      audience_filter: audienceFilter,
   182→      channel: newCampaign.channel || "all",
   183→      status: scheduledAt ? "scheduled" : "draft",
   184→      scheduled_at: scheduledAt,
   185→      sent_count: 0,
   186→      delivered_count: 0,
   187→      read_count: 0,
   188→      replied_count: 0,
   189→    });
   190→
   191→    if (!error) {
   192→      setShowCreateModal(false);
   193→      setNewCampaign({ name: "", message: "", audience: "all", channel: "all", tag: "all", minSpent: "", scheduledAt: "" });
   194→      setAudienceCount(null);
   195→      fetchCampaigns();
   196→    } else {
   197→      toast.error("Failed to create campaign: " + error.message);
   198→    }
   199→    setSaving(false);
   200→  };
   201→
   202→  const handleSendCampaign = async (campaignId) => {
   203→    if (!(await confirmAction("Send this campaign to all matching customers? This cannot be undone."))) return;
   204→    setSendingCampaignId(campaignId);
   205→    setSendResult(null);
   206→
   207→    try {
   208→      const res = await fetch("/api/campaigns/send", {
   209→        method: "POST",
   210→        headers: { "Content-Type": "application/json" },
   211→        body: JSON.stringify({ campaignId }),
   212→      });
   213→      const data = await res.json();
   214→
   215→      if (res.ok) {
   216→        setSendResult({ type: "success", message: `Campaign sent! ${data.sent} messages sent, ${data.delivered} delivered, ${data.failed} failed.` });
   217→        fetchCampaigns();
   218→      } else {
   219→        setSendResult({ type: "error", message: data.error || "Failed to send campaign" });
   220→      }
   221→    } catch (err) {
   222→      setSendResult({ type: "error", message: "Failed to send campaign: " + err.message });
   223→    }
   224→    setSendingCampaignId(null);
   225→  };
   226→
   227→  // Duplicate campaign
   228→  const handleDuplicateCampaign = async (campaign) => {
   229→    const { data: { user } } = await supabase.auth.getUser();
   230→    const { error } = await supabase.from("campaigns").insert({
   231→      account_id: user.id,
   232→      name: `${campaign.name} (Copy)`,
   233→      message_template: campaign.message_template,
   234→      audience_filter: campaign.audience_filter,
   235→      channel: campaign.channel || "all",
   236→      status: "draft",
   237→      sent_count: 0,
   238→      delivered_count: 0,
   239→      read_count: 0,
   240→      replied_count: 0,
   241→    });
   242→    if (!error) fetchCampaigns();
   243→  };
   244→
   245→  // Fetch broadcast logs for a campaign
   246→  const fetchBroadcastLogs = async (campaignId) => {
   247→    if (expandedCampaignId === campaignId) {
   248→      setExpandedCampaignId(null);
   249→      return;
   250→    }
   251→    setExpandedCampaignId(campaignId);
   252→    setLoadingLogs(true);
   253→    try {
   254→      const res = await fetch(`/api/campaigns/broadcast-logs?campaignId=${campaignId}`);
   255→      const data = await res.json();
   256→      if (data.success) {
   257→        setBroadcastLogs(data.logs || []);
   258→        setLogSummary(data.summary || null);
   259→      }
   260→    } catch (err) {
   261→      console.error("Failed to fetch broadcast logs:", err);
   262→    }
   263→    setLoadingLogs(false);
   264→  };
   265→
   266→  // Compute stats
   267→  const totalSent = campaigns.reduce((s, c) => s + (c.sent_count || 0), 0);
   268→  const totalDelivered = campaigns.reduce((s, c) => s + (c.delivered_count || 0), 0);
   269→  const totalRead = campaigns.reduce((s, c) => s + (c.read_count || 0), 0);
   270→  const totalReplied = campaigns.reduce((s, c) => s + (c.replied_count || 0), 0);
   271→  const readRate = totalDelivered > 0 ? ((totalRead / totalDelivered) * 100).toFixed(1) : "0";
   272→  const replyRate = totalDelivered > 0 ? ((totalReplied / totalDelivered) * 100).toFixed(1) : "0";
   273→
   274→  const campaignLimit = planLimits.campaigns_per_month;
   275→  const limitReached = campaignLimit !== -1 && campaignLimit === 0;
   276→
   277→  const formatDate = (d) => new Date(d).toLocaleDateString("en-US", { month: "short", day: "numeric", year: "numeric" });
   278→  const formatDateTime = (d) => new Date(d).toLocaleString("en-US", { month: "short", day: "numeric", hour: "2-digit", minute: "2-digit" });
   279→
   280→  const getStatusColor = (status) => {
   281→    switch (status) {
   282→      case "active": return "var(--accent-green)";
   283→      case "completed": return "var(--accent-secondary)";
   284→      case "scheduled": return "var(--accent-orange)";
   285→      case "draft": return "var(--text-tertiary)";
   286→      case "paused": return "var(--accent-primary-light)";
   287→      default: return "var(--text-tertiary)";
   288→    }
   289→  };
   290→
   291→  const getLogStatusColor = (status) => {
   292→    switch (status) {
   293→      case "sent": return "var(--accent-secondary)";
   294→      case "delivered": return "var(--accent-green)";
   295→      case "read": return "var(--accent-primary-light)";
   296→      case "failed": return "var(--accent-red)";
   297→      case "pending": return "var(--accent-orange)";
   298→      default: return "var(--text-tertiary)";
   299→    }
   300→  };
   301→
   302→  const getAudienceLabel = (filter) => {
   303→    if (!filter || typeof filter === "string") return filter || "All";
   304→    const parts = [];
   305→    if (filter.type && filter.type !== "all") parts.push(filter.type.charAt(0).toUpperCase() + filter.type.slice(1));
   306→    if (filter.channel && filter.channel !== "all") parts.push(filter.channel.charAt(0).toUpperCase() + filter.channel.slice(1));
   307→    if (filter.tag && filter.tag !== "all") parts.push(filter.tag);
   308→    if (filter.min_spent) parts.push(`Min ${filter.min_spent} EGP`);
   309→    return parts.length > 0 ? parts.join(" \u00b7 ") : "All Customers";
   310→  };
   311→
   312→  return (
   313→    <>
   314→      <div className="page-header">
   315→        <h1>Campaigns</h1>
   316→        <div className="page-header-actions">
   317→          {limitReached ? (
   318→            <button className="btn btn-primary" onClick={() => router.push('/dashboard/billing')} style={{ opacity: 0.7 }}>
   319→              Upgrade for Campaigns
   320→            </button>
   321→          ) : (
   322→            <button className="btn btn-primary" onClick={() => setShowCreateModal(true)}>
   323→              <Plus size={16} /> Create Campaign
   324→            </button>
   325→          )}
   326→        </div>
   327→      </div>
   328→
   329→      {/* Send result notification */}
   330→      {sendResult && (
   331→        <div style={{
   332→          padding: "var(--space-md) var(--space-lg)",
   333→          marginBottom: "var(--space-lg)",
   334→          borderRadius: "var(--radius-md)",
   335→          background: sendResult.type === "success" ? "rgba(0, 200, 83, 0.1)" : "rgba(255, 82, 82, 0.1)",
   336→          border: `1px solid ${sendResult.type === "success" ? "rgba(0, 200, 83, 0.3)" : "rgba(255, 82, 82, 0.3)"}`,
   337→          color: sendResult.type === "success" ? "var(--accent-green)" : "var(--accent-red)",
   338→          display: "flex", alignItems: "center", justifyContent: "space-between",
   339→          fontWeight: 500, fontSize: "var(--font-size-sm)",
   340→        }}>
   341→          <span>{sendResult.message}</span>
   342→          <button onClick={() => setSendResult(null)} style={{ background: "none", border: "none", color: "inherit", cursor: "pointer", padding: 4 }}>
   343→            <X size={16} />
   344→          </button>
   345→        </div>
   346→      )}
   347→
   348→      {/* Stats */}
   349→      <div className="stats-grid" style={{ marginBottom: "var(--space-xl)" }}>
   350→        {[
   351→          { label: "Total Sent", value: totalSent.toLocaleString(), icon: <Send size={18} />, color: "purple" },
   352→          { label: "Delivered", value: totalDelivered.toLocaleString(), icon: <MessageCircle size={18} />, color: "blue" },
   353→          { label: "Read Rate", value: `${readRate}%`, icon: <Eye size={18} />, color: "green" },
   354→          { label: "Reply Rate", value: `${replyRate}%`, icon: <BarChart3 size={18} />, color: "orange" },
   355→        ].map((s, i) => (
   356→          <div key={i} className="stat-card">
   357→            <div className="stat-card-header">
   358→              <div className={`stat-card-icon ${s.color}`}>{s.icon}</div>
   359→            </div>
   360→            <div className="stat-card-value" style={{ fontSize: "var(--font-size-2xl)" }}>{s.value}</div>
   361→            <div className="stat-card-label">{s.label}</div>
   362→          </div>
   363→        ))}
   364→      </div>
   365→
   366→      {/* Filters */}
   367→      <div className="filter-bar">
   368→        <div className="filter-tabs">
   369→          {["all", "draft", "scheduled", "active", "completed"].map((f) => (
   370→            <button key={f} className={`filter-tab ${filter === f ? "active" : ""}`} onClick={() => setFilter(f)}>
   371→              {f.charAt(0).toUpperCase() + f.slice(1)}
   372→            </button>
   373→          ))}
   374→        </div>
   375→      </div>
   376→
   377→      {loading ? (
   378→        <PageSkeleton showStats={false} showTable={false} />
   379→      ) : (
   380→        <div style={{ display: "flex", flexDirection: "column", gap: "var(--space-md)" }}>
   381→          {campaigns.map((campaign) => (
   382→            <div key={campaign.id} className="dashboard-panel" style={{ overflow: "visible" }}>
   383→              <div style={{ padding: "var(--space-xl)", display: "flex", alignItems: "center", justifyContent: "space-between" }}>
   384→                <div style={{ flex: 1 }}>
   385→                  <div style={{ display: "flex", alignItems: "center", gap: "var(--space-md)", marginBottom: "var(--space-sm)" }}>
   386→                    <h3 style={{ fontWeight: 700 }}>{campaign.name}</h3>
   387→                    <span style={{
   388→                      padding: "2px 10px", borderRadius: 20, fontSize: 11, fontWeight: 600,
   389→                      background: `${getStatusColor(campaign.status)}15`,
   390→                      color: getStatusColor(campaign.status),
   391→                      border: `1px solid ${getStatusColor(campaign.status)}33`,
   392→                    }}>
   393→                      {campaign.status.charAt(0).toUpperCase() + campaign.status.slice(1)}
   394→                    </span>
   395→                    {campaign.channel && campaign.channel !== "all" && (
   396→                      <span style={{
   397→                        display: "inline-flex", alignItems: "center", gap: 3,
   398→                        padding: "2px 8px", borderRadius: 12, fontSize: 10, fontWeight: 600,
   399→                        background: "var(--bg-glass)", color: "var(--text-tertiary)",
   400→                        border: "1px solid var(--border-subtle)",
   401→                      }}>
   402→                        {CHANNEL_ICON[campaign.channel]} {campaign.channel}
   403→                      </span>
   404→                    )}
   405→                  </div>
   406→                  <div style={{ display: "flex", gap: "var(--space-xl)", fontSize: "var(--font-size-sm)", color: "var(--text-tertiary)", flexWrap: "wrap" }}>
   407→                    <span><Clock size={14} style={{ display: "inline", verticalAlign: -2, marginRight: 4 }} />{formatDate(campaign.created_at)}</span>
   408→                    <span><Users size={14} style={{ display: "inline", verticalAlign: -2, marginRight: 4 }} />{getAudienceLabel(campaign.audience_filter)}</span>
   409→                    {campaign.scheduled_at && (
   410→                      <span><Calendar size={14} style={{ display: "inline", verticalAlign: -2, marginRight: 4 }} />Scheduled: {formatDateTime(campaign.scheduled_at)}</span>
   411→                    )}
   412→                  </div>
   413→                  {/* Message preview */}
   414→                  <div style={{
   415→                    marginTop: "var(--space-sm)", padding: "var(--space-sm) var(--space-md)",
   416→                    background: "var(--bg-glass)", borderRadius: "var(--radius-md)",
   417→                    fontSize: "var(--font-size-sm)", color: "var(--text-secondary)",
   418→                    border: "1px solid var(--border-subtle)", maxWidth: 500,
   419→                    whiteSpace: "nowrap", overflow: "hidden", textOverflow: "ellipsis",
   420→                  }}>
   421→                    {campaign.message_template || "No message template"}
   422→                  </div>
   423→                </div>
   424→
   425→                {campaign.sent_count > 0 && (
   426→                  <div style={{ display: "flex", gap: "var(--space-2xl)", textAlign: "center" }}>
   427→                    {[
   428→                      { label: "Sent", value: campaign.sent_count.toLocaleString() },
   429→                      { label: "Delivered", value: campaign.delivered_count.toLocaleString() },
   430→                      { label: "Read", value: campaign.read_count.toLocaleString() },
   431→                      { label: "Replied", value: campaign.replied_count.toLocaleString() },
   432→                    ].map((m, i) => (
   433→                      <div key={i}>
   434→                        <div style={{ fontWeight: 700, fontSize: "var(--font-size-lg)" }}>{m.value}</div>
   435→                        <div style={{ fontSize: "var(--font-size-xs)", color: "var(--text-tertiary)" }}>{m.label}</div>
   436→                      </div>
   437→                    ))}
   438→                  </div>
   439→                )}
   440→
   441→                <div style={{ display: "flex", gap: "var(--space-xs)", marginLeft: "var(--space-xl)" }}>
   442→                  {/* View broadcast logs */}
   443→                  {campaign.sent_count > 0 && (
   444→                    <button
   445→                      className="topbar-btn"
   446→                      title="View delivery details"
   447→                      onClick={() => fetchBroadcastLogs(campaign.id)}
   448→                    >
   449→                      {expandedCampaignId === campaign.id ? <ChevronUp size={16} /> : <Eye size={16} />}
   450→                    </button>
   451→                  )}
   452→                  {campaign.status === "active" && (
   453→                    <button className="topbar-btn" title="Pause" onClick={() => handleStatusChange(campaign.id, "paused")}><Pause size={16} /></button>
   454→                  )}
   455→                  {(campaign.status === "draft" || campaign.status === "paused") && (
   456→                    <button
   457→                      className="topbar-btn"
   458→                      title="Send Now"
   459→                      onClick={() => handleSendCampaign(campaign.id)}
   460→                      disabled={sendingCampaignId === campaign.id}
   461→                      style={{ color: "var(--accent-green)" }}
   462→                    >
   463→                      {sendingCampaignId === campaign.id ? <Loader2 size={16} className="spin" /> : <Zap size={16} />}
   464→                    </button>
   465→                  )}
   466→                  {campaign.status === "draft" && (
   467→                    <button className="topbar-btn" title="Activate" onClick={() => handleStatusChange(campaign.id, "active")}><Send size={16} /></button>
   468→                  )}
   469→                  {/* Duplicate campaign */}
   470→                  <button className="topbar-btn" title="Duplicate" onClick={() => handleDuplicateCampaign(campaign)}><Copy size={16} /></button>
   471→                  <button className="topbar-btn" title="Delete" onClick={() => handleDelete(campaign.id)}><Trash2 size={16} /></button>
   472→                </div>
   473→              </div>
   474→
   475→              {/* Broadcast logs expanded section */}
   476→              {expandedCampaignId === campaign.id && (
   477→                <div style={{
   478→                  borderTop: "1px solid var(--border-subtle)",
   479→                  padding: "var(--space-lg) var(--space-xl)",
   480→                  background: "var(--bg-glass)",
   481→                }}>
   482→                  {loadingLogs ? (
   483→                    <div style={{ textAlign: "center", padding: "var(--space-lg)", color: "var(--text-tertiary)" }}>

   551→                  )}
   552→                </div>
   553→              )}
   554→            </div>
   555→          ))}
   556→          {campaigns.length === 0 && (
   557→            <div style={{ padding: "var(--space-3xl)", textAlign: "center", color: "var(--text-tertiary)" }}>

   633→                                if (cond.field === "total_spent" && cond.operator === "greater_than") updates.minSpent = String(cond.value);

   636→                            }

   640→                        <option value="">No Segment</option>
   641→                        {savedSegments.map((seg) => (

   668→                        <option value="facebook">Facebook Only</option>
   669→                        <option value="whatsapp">WhatsApp Only</option>
   670→                      </select>
   671→                    </div>
   672→                    <div className="form-group" style={{ marginBottom: 0 }}>

   693→                        min="0"
   694→                      />
   695→                    </div>

   698→                  <div style={{
   699→                    marginTop: "var(--space-md)", padding: "var(--space-sm) var(--space-md)",
   700→                    background: "rgba(108, 92, 231, 0.08)", borderRadius: "var(--radius-sm)",
   701→                    fontSize: "var(--font-size-sm)", color: "var(--accent-primary-light)",

   707→                </div>
   708→
   709→                {/* Schedule */}

   711→                  <label className="form-label" style={{ display: "flex", alignItems: "center", gap: 6 }}>
   712→                    <Calendar size={14} /> Schedule (optional)
   713→                  </label>

   716→                    className="form-input"
