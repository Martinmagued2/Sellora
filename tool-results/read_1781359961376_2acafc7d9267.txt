   316→   365→   365→      providerModels.push({ name: 'openai', model: openai('gpt-4o-mini') });
   317→   366→   366→    }
   318→   367→   367→
   319→   368→   368→    // NVIDIA NIM
   320→   369→   369→    if (process.env.NVIDIA_API_KEY) {
   321→   370→   370→      const nvidia = createOpenAI({
   322→   371→   371→        apiKey: process.env.NVIDIA_API_KEY,
   323→   372→   372→        baseURL: process.env.NVIDIA_BASE_URL || 'https://integrate.api.nvidia.com/v1',
   324→   373→   373→        compatibility: 'compatible',
   325→   374→   374→      });
   326→   375→   375→      providerModels.push({ name: 'nvidia-llama33-70b', model: nvidia('meta/llama-3.3-70b-instruct') });
   327→   376→   376→      providerModels.push({ name: 'nvidia-nemotron-70b', model: nvidia('nvidia/llama-3.1-nemotron-70b-instruct') });
   328→   377→   377→      providerModels.push({ name: 'nvidia-deepseek-r1', model: nvidia('deepseek-ai/deepseek-r1') });
   329→   378→   378→      providerModels.push({ name: 'nvidia-mistral-large', model: nvidia('mistralai/mistral-large-2-instruct') });
   330→   379→   379→    }
   331→   380→   380→
   332→   381→   381→    if (providerModels.length === 0) {
   333→   382→   382→      return Response.json({ error: 'AI is not configured. Please add GROQ_API_KEY or GOOGLE_GENERATIVE_AI_API_KEY to your .env.local file.' }, { status: 500 });
   334→   383→   383→    }
   335→   384→   384→
   336→   385→   385→    const tools = createCopilotTools(user.id);
   337→   386→   386→
   338→   387→   387→    // ─── Use streamText for proper streaming with useChat ───
   339→   388→   388→    // streamText produces the correct stream format that the useChat hook
   340→   389→   389→    // can parse, including both tool calls AND text responses.
   341→   390→   390→    // Error handling: If streamText fails at the start (rate limit, auth error),
   342→   391→   391→    // it throws before we return the response, so we can fall back to the next provider.
   343→   392→   392→    // Mid-stream errors cannot be caught (same trade-off as /api/agent/route.js),
   344→   393→   393→    // but initial errors (most common) are handled properly.
   345→   394→   394→
   346→   395→   395→    let groqRateLimited = false;
   347→   396→   396→    let lastError = null;
   348→   397→   397→    let lastErrorType = 'unknown'; // Track error type for better messages
   349→   398→   398→
   350→   399→   399→    // Attempt 1: Try each provider with tools (streaming)
   351→   400→   400→    for (const providerEntry of providerModels) {
   352→   401→   401→      if (groqRateLimited && providerEntry.name.startsWith('groq-')) {
   353→   402→   402→        console.warn(`[Agent] Skipping ${providerEntry.name} because Groq rate limit was already hit`);
   354→   403→   403→        continue;
   355→   404→   404→      }
   356→   405→   405→
   357→   406→   406→      try {
   358→   407→   407→        const result = await streamText({
   359→   408→   408→          model: providerEntry.model,
   360→   409→   409→          maxSteps: 5,
   361→   410→   410→          temperature: 0.2,
   362→   411→   411→          system: systemPrompt,
   363→   412→   412→          messages: coreMessages,
   364→   413→   413→          tools,
   365→   414→   414→        });
   366→   415→   415→
   367→   416→   416→        console.log(`[Agent] ${providerEntry.name} stream started successfully`);
   368→   417→   417→        return result.toUIMessageStreamResponse();
   369→   418→   418→      } catch (providerError) {
   370→   419→   419→        lastError = providerError;
   371→   420→   420→        const errMsg = providerError?.message || '';
   372→   421→   421→        console.warn(`[Agent] ${providerEntry.name} failed:`, errMsg.substring(0, 200));
   373→   422→   422→
   374→   423→   423→        // Detect Groq rate limit — mark all Groq providers as unavailable
   375→   424→   424→        if (errMsg.includes('Rate limit') && providerEntry.name.startsWith('groq-')) {
   376→   425→   425→          groqRateLimited = true;
   377→   426→   426→          lastErrorType = 'rate_limit';
   378→   427→   427→          console.warn(`[Agent] Groq rate limit detected, skipping remaining Groq providers`);
   379→   428→   428→        }
   380→   429→   429→        // Detect Groq function calling failures — these are transient, try next provider
   381→   430→   430→        if (errMsg.includes('Failed to call a function') || errMsg.includes('invalid_request_error')) {
   382→   431→   431→          lastErrorType = 'function_error';
   383→   432→   432→          console.warn(`[Agent] ${providerEntry.name} had function calling error, trying next provider`);
   384→   433→   433→        }
   385→   434→   434→        // Detect auth errors
   386→   435→   435→        if (errMsg.includes('Invalid API Key') || errMsg.includes('Unauthorized') || errMsg.includes('authentication')) {
   387→   436→   436→          lastErrorType = 'auth_error';
   388→   437→   437→        }
   389→   438→   438→        // Detect server errors / overload
   390→   439→   439→        if (errMsg.includes('overloaded') || errMsg.includes('503') || errMsg.includes('500')) {
   391→   440→   440→          lastErrorType = 'server_error';
   392→   441→   441→        }
   393→   442→   442→      }
   394→   443→   443→    }
   395→   444→   444→
   396→   445→   445→    // Attempt 2: Fallback — stream WITHOUT tools
   397→   446→   446→    console.warn("[Agent] All providers with tools failed, trying without tools...");
   398→   447→   447→    for (const providerEntry of providerModels) {
   399→   448→   448→      if (groqRateLimited && providerEntry.name.startsWith('groq-')) {
   400→   449→   449→        continue;
   401→   450→   450→      }
   402→   451→   451→
   403→   452→   452→      try {
   404→   453→   453→        const result = await streamText({
   405→   454→   454→          model: providerEntry.model,
   406→   455→   455→          maxSteps: 1,
   407→   456→   456→          temperature: 0.2,
   408→   457→   457→          system: systemPrompt,
   409→   458→   458→          messages: coreMessages,
   410→   459→   459→        });
   411→   460→   460→
   412→   461→   461→        console.log(`[Agent] ${providerEntry.name} stream started without tools`);
   413→   462→   462→        return result.toUIMessageStreamResponse();
   414→   463→   463→      } catch (providerError) {
   415→   464→   464→        console.warn(`[Agent] ${providerEntry.name} without tools also failed:`, providerError?.message?.substring(0, 120) || providerError);
   416→   465→   465→      }
   417→   466→   466→    }
   418→   467→   467→
   419→   468→   468→    // Return a user-friendly error message based on the error type
   420→   469→   469→    let userMessage;
   421→   470→   470→    switch (lastErrorType) {
   422→   471→   471→      case 'rate_limit':
   423→   472→   472→        userMessage = 'Daily AI usage limit reached. Please try again in a few minutes when the limit resets.';
   424→   473→   473→        break;
   425→   474→   474→      case 'auth_error':
   426→   475→   475→        userMessage = 'AI service configuration issue. Please check your API keys.';
   427→   476→   476→        break;
   428→   477→   477→      case 'server_error':
   429→   478→   478→        userMessage = 'AI service is temporarily overloaded. Please try again shortly.';
   430→   479→   479→        break;
   431→   480→   480→      case 'function_error':
   432→   481→   481→        userMessage = 'AI had trouble processing the request. Please try again or rephrase your message.';
   433→   482→   482→        break;
   434→   483→   483→      default:
   435→   484→   484→        userMessage = 'All AI providers are currently unavailable. Please try again in a few minutes.';
   436→   485→   485→        break;
   437→   486→   486→    }
   438→   487→   487→
   439→   488→   488→    return Response.json({ error: userMessage }, { status: 500 });
   440→   489→   489→  } catch (error) {
   441→   490→   490→    console.error("Agent API Error:", error);
   442→   491→   491→    return Response.json({ error: error.message || "Something went wrong." }, { status: 500 });
   443→   492→   492→  }
   444→   493→   493→}
