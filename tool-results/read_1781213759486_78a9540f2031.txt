    56→    get_sales_report: tool({
    57→      description: "Generate a detailed sales/income report for the store. Includes revenue breakdown, order stats, top-selling products, and trends.     58→      inputSchema: z.object({
    59→        period: z.string().describe("The time period for the report. Must be one of: today, week, month, quarter, year"),
    60→      }),
    61→      execute: async ({ period }) => {
    62→        // Normalize period string to valid value
    63→        const periodMap = { today: 'today', week: 'week', month: 'month', quarterly: 'quarter', quarter: 'quarter', year: 'year', yearly: 'year', annual: 'year', daily: 'today', weekly: 'week', monthly: 'month' };

    65→        const now = new Date();

    67→
    68→        switch (normalizedPeriod) {
    69→          case "today":
    70→            startDate = new Date(now.getFullYear(), now.getMonth(), now.getDate());
    71→            break;
    72→          case "week":
    73→            startDate = new Date(now);
    74→            startDate.setDate(startDate.getDate() - 7);
    75→            break;
    76→          case "month":
    77→            startDate = new Date(now);
    78→            startDate.setDate(startDate.getDate() - 30);
    79→            break;
    80→          case "quarter":
    81→            startDate = new Date(now);
    82→            startDate.setDate(startDate.getDate() - 90);
    83→            break;
    84→          case "year":
    85→            startDate = new Date(now);
    86→            startDate.setDate(startDate.getDate() - 365);
    87→            break;
    88→        }
    89→
    90→        // Fetch orders for the period
    91→        const { data: orders, error: ordersError } = await supabase
    92→          .from("orders")
    93→          .select("id, total, status, created_at, items, payment_method")
    94→          .eq("account_id", accountId)
    95→          .gte("created_at", startDate.toISOString());
    96→
    97→        if (ordersError) return { success: false, error: "Failed to fetch orders for report" };
    98→
    99→        // Fetch products for inventory analysis
   100→        const { data: products } = await supabase
   101→          .from("products")
   102→          .select("id, name, price, stock, category, status")
   103→          .eq("account_id", accountId);
   104→
   105→        // Calculate metrics
   106→        const totalRevenue = orders.reduce((sum, o) => sum + (o.total || 0), 0);
   107→        const completedRevenue = orders.filter(o => o.status === "delivered" || o.status === "confirmed").reduce((sum, o) => sum + (o.total || 0), 0);
   108→        const pendingRevenue = orders.filter(o => o.status === "pending").reduce((sum, o) => sum + (o.total || 0), 0);
   109→        const totalOrders = orders.length;
   110→        const completedOrders = orders.filter(o => o.status === "delivered" || o.status === "confirmed").length;
   111→        const pendingOrders = orders.filter(o => o.status === "pending").length;
   112→        const cancelledOrders = orders.filter(o => o.status === "cancelled").length;
   113→        const avgOrderValue = totalOrders > 0 ? Math.round(totalRevenue / totalOrders) : 0;
   114→
   115→        // Top selling products from order items
   116→        const productSales = {};
   117→        for (const order of orders) {
   118→          if (order.items && Array.isArray(order.items)) {
   119→            for (const item of order.items) {
   120→              const name = item.name || "Unknown";
   121→              if (!productSales[name]) productSales[name] = { qty: 0, revenue: 0 };
   122→              productSales[name].qty += item.qty || 0;
   123→              productSales[name].revenue += (item.price || 0) * (item.qty || 0);
   124→            }
   125→          }
   126→        }
   127→        const topProducts = Object.entries(productSales)
   128→          .sort(([, a], [, b]) => b.revenue - a.revenue)
   129→          .slice(0, 5)
   130→          .map(([name, data]) => ({ name, ...data }));
   131→
   132→        // Payment method breakdown
   133→        const paymentBreakdown = {};
   134→        for (const order of orders) {
   135→          const method = order.payment_method || "unknown";
   136→          if (!paymentBreakdown[method]) paymentBreakdown[method] = { count: 0, total: 0 };
   137→          paymentBreakdown[method].count++;
   138→          paymentBreakdown[method].total += order.total || 0;
   139→        }
   140→
   141→        // Inventory summary
   142→        const activeProducts = products?.filter(p => p.status === "active") || [];
   143→        const outOfStock = activeProducts.filter(p => p.stock <= 0).length;
   144→        const lowStock = activeProducts.filter(p => p.stock > 0 && p.stock <= 5).length;
   145→        const totalInventoryValue = activeProducts.reduce((sum, p) => sum + (p.price * p.stock), 0);
   146→
   147→        return {
   148→          success: true,
   149→          period: normalizedPeriod,
   150→          startDate: startDate.toISOString(),
   151→          generatedAt: now.toISOString(),
   152→          revenue: {
   153→            total: totalRevenue,
   154→            completed: completedRevenue,
   155→            pending: pendingRevenue,
   156→          },

   158→            total: totalOrders,
   159→            completed: completedOrders,
   160→            pending: pendingOrders,
   161→            cancelled: cancelledOrders,
   162→            avgValue: avgOrderValue,
   163→          },
   164→          topProducts,
   165→          paymentBreakdown,
   166→          inventory: {
   167→            activeProducts: activeProducts.length,
   168→            outOfStock,
   169→            lowStock,
   170→            totalInventoryValue,
   171→          },
   172→          _action: { type: "navigate", path: "/dashboard/analytics", label: "View Full Analytics" },
   173→        };
   174→      },
