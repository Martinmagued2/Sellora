     4→import {

    25→function formatTimeAgo(dateStr) {
    26→  if (!dateStr) return "N/A";
    27→  const diff = Date.now() - new Date(dateStr).getTime();
    28→  const minutes = Math.floor(diff / 60000);
    29→  const hours = Math.floor(minutes / 60);
    30→  const days = Math.floor(hours / 24);
    31→
    32→  if (days > 0) return `${days}d ago`;
    33→  if (hours > 0) return `${hours}h ago`;
    34→  if (minutes > 0) return `${minutes}m ago`;
    35→  return "Just now";
    36→}
    37→
    38→function formatCurrency(value) {
    39→  return `${(parseFloat(value) || 0).toLocaleString()} EGP`;
    40→}

    43→  const [carts, setCarts] = useState([]);
    44→  const [stats, setStats] = useState({
    45→    totalAbandonedValue: 0,

   116→          account_id: user.id,

   118→          include_discount: false,

   148→        method: "POST",
   149→        headers: { "Content-Type": "application/json" },
   150→        body: JSON.stringify({

   303→              <Package size={20} />

   307→          <div className="stat-card-label">Recovered Revenue</div>
   308→        </div>
   309→      </div>

   311→      {/* Filter Bar */}

   313→        <div className="filter-tabs">
   314→          {[

   316→            { key: "abandoned", label: "Abandoned" },
   317→            { key: "reminded", label: "Reminded" },
   318→            { key: "recovered", label: "Recovered" },

   321→              {f.label}

   324→        </div>
   325→        <div className="filter-search">
   326→          <Search size={14} />
   327→          <input
   328→            type="text"
   329→            placeholder="Search by customer or items..."
   330→            value={search}

   357→                  <th>Value</th>
   358→                  <th>Abandoned</th>

   377→                            alignItems: "center", justifyContent: "center",
   378→                            fontWeight: 700, fontSize: 12, flexShrink: 0,

   380→                            {getInitials(cart.customer?.name)}

   406→                      </td>

   631→
   632→              {/* Coupon Code */}
   633→              {selectedCart.coupon_code && (

   644→                  </div>
   645→                </div>
   646→              )}
   647→            </div>
   648→
   649→            {/* Modal Footer - Quick Actions */}
   650→            <div className="modal-footer" style={{ flexWrap: "wrap", gap: "var(--space-sm)" }}>

   669→              <button className="btn btn-secondary" onClick={() => setSelectedCart(null)}>Close</button>

   671→          </div>
