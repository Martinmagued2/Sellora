     1→"use client";
     2→
     3→import { useState, useEffect } from "react";
     4→import Link from "next/link";
     5→import {
     6→  MessageCircle, Clock, ToggleLeft, ToggleRight, AlertCircle,
     7→  HelpCircle, Zap, Send, Plus, Edit, Trash2, Loader2, Megaphone,
     8→  Globe, Smartphone, Save, Check, ShoppingCart, ChevronRight,
     9→} from "lucide-react";
    10→import { createClient } from "@/lib/supabase/client";
    11→import { SAFE_ACCOUNT_FIELDS } from "@/lib/safe-fields";
    12→import { useToast } from "../components/ToastProvider";
    13→import { useConfirm } from "../components/ConfirmProvider";
    14→
    15→export default function AutomationPage() {
    16→  const toast = useToast();
    17→  // Account / automation state
    18→  
    19→  const confirmAction = useConfirm();
    20→  const [loading, setLoading] = useState(true);
    21→  const [saving, setSaving] = useState(false);
    22→  const [saved, setSaved] = useState(false);
    23→
    24→  const [account, setAccount] = useState({});
    25→  const [autoGreeting, setAutoGreeting] = useState(false);
    26→  const [autoGreetingMessage, setAutoGreetingMessage] = useState("");
    27→  const [greetingPerChannel, setGreetingPerChannel] = useState(false);
    28→  const [instagramGreeting, setInstagramGreeting] = useState("");
    29→  const [facebookGreeting, setFacebookGreeting] = useState("");
    30→  const [whatsappGreeting, setWhatsappGreeting] = useState("");
    31→  const [greetingDelaySeconds, setGreetingDelaySeconds] = useState(0);
    32→  const [autoFollowUp, setAutoFollowUp] = useState(false);
    33→  const [sendingFollowUps, setSendingFollowUps] = useState(false);
    34→
    35→  // Quick Replies state
    36→  const [quickReplies, setQuickReplies] = useState([]);
    37→  const [showAddQuickReply, setShowAddQuickReply] = useState(false);
    38→  const [newQuickReply, setNewQuickReply] = useState({ title: "", content: "", category: "General", shortcut: "" });
    39→  const [quickReplySaving, setQuickReplySaving] = useState(false);
    40→  const [editingQuickReply, setEditingQuickReply] = useState(null);
    41→
    42→  // FAQ count
    43→  const [faqCount, setFaqCount] = useState(0);
    44→
    45→  // Abandoned Cart Recovery state
    46→  const [abandonedCartEnabled, setAbandonedCartEnabled] = useState(false);
    47→  const [abandonedCartHours, setAbandonedCartHours] = useState(2);
    48→  const [abandonedCartAutoReminder, setAbandonedCartAutoReminder] = useState(false);
    49→  const [abandonedCartReminderHours, setAbandonedCartReminderHours] = useState(1);
    50→  const [abandonedCartAutoSecondReminder, setAbandonedCartAutoSecondReminder] = useState(false);
    51→  const [abandonedCartSecondReminderHours, setAbandonedCartSecondReminderHours] = useState(24);
    52→  const [abandonedCartDiscountPercent, setAbandonedCartDiscountPercent] = useState(10);
    53→
    54→  // Broadcast state (Phase 2)
    55→  const [broadcastAudience, setBroadcastAudience] = useState("all");
    56→  const [broadcastChannel, setBroadcastChannel] = useState("instagram");
    57→  const [broadcastMessage, setBroadcastMessage] = useState("");
    58→  const [broadcastSending, setBroadcastSending] = useState(false);
    59→  const [broadcastResult, setBroadcastResult] = useState(null);
    60→  const [showBroadcast, setShowBroadcast] = useState(false);
    61→
    62→  // Lazy supabase client - only created on the client side
    63→  const getSupabase = () => createClient();
    64→
    65→  useEffect(() => {
    66→    const load = async () => {
    67→      try {
    68→        const supabase = getSupabase();
    69→        const { data: { user } } = await supabase.auth.getUser();
    70→        if (!user) return;
    71→
    72→        const { data } = await supabase.from("accounts").select(SAFE_ACCOUNT_FIELDS).eq("id", user.id).single();
    73→        if (data) {
    74→          setAccount(data);
    75→          setAutoGreeting(data.auto_greeting || false);
    76→          setAutoGreetingMessage(data.auto_greeting_message || "");
    77→          setGreetingPerChannel(data.greeting_per_channel || false);
    78→          setInstagramGreeting(data.instagram_greeting || "");
    79→          setFacebookGreeting(data.facebook_greeting || "");
    80→          setWhatsappGreeting(data.whatsapp_greeting || "");
    81→          setGreetingDelaySeconds(data.greeting_delay_seconds || 0);
    82→          setAutoFollowUp(data.auto_follow_up_enabled || false);
    83→          setAbandonedCartEnabled(data.abandoned_cart_enabled || false);
    84→          setAbandonedCartHours(data.abandoned_cart_hours || 2);
    85→          setAbandonedCartAutoReminder(data.abandoned_cart_auto_reminder || false);
    86→          setAbandonedCartReminderHours(data.abandoned_cart_reminder_hours || 1);
    87→          setAbandonedCartAutoSecondReminder(data.abandoned_cart_auto_second_reminder || false);
    88→          setAbandonedCartSecondReminderHours(data.abandoned_cart_second_reminder_hours || 24);
    89→          setAbandonedCartDiscountPercent(data.abandoned_cart_discount_percent || 10);
    90→        }
    91→
    92→        // Fetch quick replies
    93→        const { data: qrData } = await supabase.from("quick_replies").select("*").eq("account_id", user.id).order("created_at");
    94→        if (qrData) setQuickReplies(qrData);
    95→
    96→        // Fetch FAQ count
    97→        const { count } = await supabase.from("faqs").select("*", { count: "exact", head: true }).eq("account_id", user.id);
    98→        setFaqCount(count || 0);
    99→      } catch (err) {
   100→        console.error("Automation page load error:", err);
   101→      } finally {
   102→        setLoading(false);
   103→      }
   104→    };
   105→    load();
   106→  }, []);
   107→
   108→  // Helper: update account field
   109→  const updateAccount = async (field, value) => {
   110→    const supabase = getSupabase();
   111→    const { data: { user } } = await supabase.auth.getUser();
   112→    if (!user) return;
   113→    await supabase.from("accounts").update({ [field]: value }).eq("id", user.id);
   114→  };
   115→
   116→  const handleSave = async () => {
   117→    setSaving(true);
   118→    try {
   119→      const supabase = getSupabase();
   120→      const { data: { user } } = await supabase.auth.getUser();
   121→      if (!user) return;
   122→      await supabase.from("accounts").update({
   123→        auto_greeting: autoGreeting,
   124→        auto_greeting_message: autoGreetingMessage,
   125→        greeting_per_channel: greetingPerChannel,
   126→        instagram_greeting: instagramGreeting,
   127→        facebook_greeting: facebookGreeting,
   128→        whatsapp_greeting: whatsappGreeting,
   129→        greeting_delay_seconds: greetingDelaySeconds,
   130→        auto_follow_up_enabled: autoFollowUp,
   131→        abandoned_cart_enabled: abandonedCartEnabled,
   132→        abandoned_cart_hours: abandonedCartHours,
   133→        abandoned_cart_auto_reminder: abandonedCartAutoReminder,
   134→        abandoned_cart_reminder_hours: abandonedCartReminderHours,
   135→        abandoned_cart_auto_second_reminder: abandonedCartAutoSecondReminder,
   136→        abandoned_cart_second_reminder_hours: abandonedCartSecondReminderHours,
   137→        abandoned_cart_discount_percent: abandonedCartDiscountPercent,
   138→      }).eq("id", user.id);
   139→      setSaved(true);
   140→      setTimeout(() => setSaved(false), 2000);
   141→    } catch (err) {
   142→      console.error("Save error:", err);
   143→    }
   144→    setSaving(false);
   145→  };
   146→
   147→  const handleSendFollowUps = async () => {
   148→    setSendingFollowUps(true);
   149→    try {
   150→      const supabase = getSupabase();
   151→      const { data: { user } } = await supabase.auth.getUser();
   152→      const res = await fetch("/api/automation/follow-up", {
   153→        method: "POST",
   154→        headers: { "Content-Type": "application/json" },
   155→        body: JSON.stringify({ account_id: user.id }),
   156→      });
   157→      const data = await res.json();
   158→      toast.info(data.message || `Sent ${data.sent || 0} follow-up messages`);
   159→    } catch (err) {
   160→      toast.error("Failed to send follow-ups: " + err.message);
   161→    }
   162→    setSendingFollowUps(false);
   163→  };
   164→
   165→  const handleQuickReplySave = async () => {
   166→    setQuickReplySaving(true);
   167→    try {
   168→      if (editingQuickReply) {
   169→        const res = await fetch("/api/quick-replies", {
   170→          method: "PUT",
   171→          headers: { "Content-Type": "application/json" },
   172→          body: JSON.stringify({ id: editingQuickReply.id, ...newQuickReply }),
   173→        });
   174→        const data = await res.json();
   175→        if (data.success) {
   176→          setQuickReplies((prev) => prev.map(qr => qr.id === editingQuickReply.id ? data.quickReply : qr));
   177→        }
   178→      } else {
   179→        const res = await fetch("/api/quick-replies", {
   180→          method: "POST",
   181→          headers: { "Content-Type": "application/json" },
   182→          body: JSON.stringify(newQuickReply),
   183→        });
   184→        const data = await res.json();
   185→        if (data.success) {
   186→          setQuickReplies((prev) => [...prev, data.quickReply]);
   187→        }
   188→      }
   189→      setShowAddQuickReply(false);
   190→      setEditingQuickReply(null);
   191→      setNewQuickReply({ title: "", content: "", category: "General", shortcut: "" });
   192→    } catch (err) {
   193→      console.error("Quick reply save error:", err);
   194→    }
   195→    setQuickReplySaving(false);
   196→  };
   197→
   198→  const handleQuickReplyDelete = async (id) => {
   199→    if (!(await confirmAction("Delete this template?"))) return;
   200→    await fetch(`/api/quick-replies?id=${id}`, { method: "DELETE" });
   201→    setQuickReplies((prev) => prev.filter(q => q.id !== id));
   202→  };
   203→
   204→  const handleBroadcast = async () => {
   205→    setBroadcastSending(true);
   206→    setBroadcastResult(null);
   207→    try {
   208→      const res = await fetch("/api/automation/broadcast", {
   209→        method: "POST",
   210→        headers: { "Content-Type": "application/json" },
   211→        body: JSON.stringify({
   212→          audience: broadcastAudience,
   213→          channel: broadcastChannel,
   214→          message: broadcastMessage,
   215→        }),
   216→      });
   217→      const data = await res.json();
   218→      setBroadcastResult(data);
   219→    } catch (err) {
   220→      setBroadcastResult({ error: "Failed to send broadcast: " + err.message });
   221→    }
   222→    setBroadcastSending(false);
   223→  };
   224→
   225→  if (loading) {
   226→    return (
   227→      <div style={{ display: "flex", alignItems: "center", justifyContent: "center", minHeight: "60vh" }}>
   228→        <Loader2 size={32} className="spin" style={{ color: "var(--accent-primary)" }} />
   229→      </div>
   230→    );
   231→  }
   232→
   233→  return (
   234→    <>
   235→      {/* Page Header */}
   236→      <div className="page-header">
   237→        <h1>Automation</h1>
   238→        <div className="page-header-actions">
   239→          <button className="btn btn-primary" onClick={handleSave} disabled={saving}>
   240→            {saved ? <><Check size={16} /> Saved!</> : saving ? <><Loader2 size={16} className="spin" /> Saving...</> : <><Save size={16} /> Save Changes</>}
   241→          </button>
   242→        </div>
   243→      </div>
   244→
   245→      {/* Section 1: Auto-Greeting */}
   246→      <div className="dashboard-panel" style={{ marginBottom: "var(--space-xl)" }}>
   247→        <div className="dashboard-panel-header">
   248→          <h3 style={{ display: "flex", alignItems: "center", gap: 8 }}>
   249→            <MessageCircle size={18} style={{ color: "var(--accent-primary-light)" }} />
   250→            Auto-Greeting
   251→          </h3>
   252→        </div>
   253→        <div className="dashboard-panel-body" style={{ padding: "var(--space-xl)" }}>
   254→          {/* Toggle */}
   255→          <div style={{
   256→            display: "flex", alignItems: "center", justifyContent: "space-between",
   257→            padding: "var(--space-lg)", background: "var(--bg-glass)",
   258→            borderRadius: "var(--radius-md)", marginBottom: "var(--space-lg)",
   259→          }}>
   260→            <div>
   261→              <div style={{ fontWeight: 600, marginBottom: 4 }}>Auto-Greeting</div>
   262→              <div style={{ fontSize: "var(--font-size-sm)", color: "var(--text-tertiary)" }}>
   263→                Automatically send a welcome message to new customers on their first message
   264→              </div>
   265→            </div>
   266→            <div
   267→              style={{ color: autoGreeting ? "var(--accent-green)" : "var(--text-tertiary)", cursor: "pointer" }}
   268→              onClick={() => {
   269→                const newVal = !autoGreeting;
   270→                setAutoGreeting(newVal);
   271→              }}
   272→            >
   273→              {autoGreeting ? <ToggleRight size={36} /> : <ToggleLeft size={36} />}
   274→            </div>
   275→          </div>
   276→
   277→          {/* Greeting Configuration (shown when enabled) */}
   278→          {autoGreeting && (
   279→            <>
   280→              {/* Default Greeting Message */}
   281→              <div className="form-group" style={{ marginBottom: "var(--space-lg)" }}>
   282→                <label className="form-label">Default Greeting Message</label>
   283→                <textarea
   284→                  className="form-input form-textarea"
   285→                  value={autoGreetingMessage}
   286→                  onChange={(e) => setAutoGreetingMessage(e.target.value)}
   287→                  rows={3}
   288→                  placeholder="Hi! Welcome to {business_name} 👋 How can I help you today?"
   289→                />
   290→                <p style={{ fontSize: "var(--font-size-xs)", color: "var(--text-tertiary)", marginTop: 4 }}>
   291→                  Use {"{business_name}"} for your store name, {"{name}"} for the customer&apos;s name.
   292→                </p>
   293→              </div>
   294→
   295→              {/* Greeting Delay */}
   296→              <div className="form-group" style={{ marginBottom: "var(--space-lg)" }}>
   297→                <label className="form-label">Greeting Delay (seconds)</label>
   298→                <div style={{ display: "flex", alignItems: "center", gap: "var(--space-md)" }}>
   299→                  <input
   300→                    type="range"
   301→                    min="0"
   302→                    max="30"
   303→                    value={greetingDelaySeconds}
   304→                    onChange={(e) => setGreetingDelaySeconds(Number(e.target.value))}
   305→                    style={{ flex: 1, accentColor: "var(--accent-primary)" }}
   306→                  />
   307→                  <span style={{
   308→                    minWidth: 50, textAlign: "center", fontWeight: 600,
   309→                    fontSize: "var(--font-size-sm)", color: "var(--accent-primary)",
   310→                  }}>
   311→                    {greetingDelaySeconds}s
   312→                  </span>
   313→                </div>
   314→                <p style={{ fontSize: "var(--font-size-xs)", color: "var(--text-tertiary)", marginTop: 4 }}>
   315→                  Wait before sending (0 = instant, max 30s)
   316→                </p>
   317→              </div>
   318→
   319→              {/* Per-Channel Greeting Toggle */}
   320→              <div style={{
   321→                display: "flex", alignItems: "center", justifyContent: "space-between",
   322→                padding: "var(--space-md)", background: "var(--bg-glass)",
   323→                borderRadius: "var(--radius-md)", marginBottom: "var(--space-md)",
   324→              }}>
   325→                <div>
   326→                  <div style={{ fontWeight: 600, fontSize: "var(--font-size-sm)" }}>Custom Greetings Per Channel</div>
   327→                  <div style={{ fontSize: "var(--font-size-xs)", color: "var(--text-tertiary)", marginTop: 2 }}>
   328→                    Set different greeting messages for Instagram, Facebook, and WhatsApp
   329→                  </div>
   330→                </div>
   331→                <div
   332→                  style={{ color: greetingPerChannel ? "var(--accent-green)" : "var(--text-tertiary)", cursor: "pointer" }}
   333→                  onClick={() => setGreetingPerChannel(!greetingPerChannel)}
   334→                >
   335→                  {greetingPerChannel ? <ToggleRight size={32} /> : <ToggleLeft size={32} />}
   336→                </div>
   337→              </div>
   338→
   339→              {/* Per-Channel Greeting Messages */}
   340→              {greetingPerChannel && (
   341→                <div style={{ display: "grid", gridTemplateColumns: "1fr 1fr 1fr", gap: "var(--space-md)", marginBottom: "var(--space-xl)" }}>
   342→                  <div className="form-group" style={{ marginBottom: 0 }}>
   343→                    <label className="form-label" style={{ display: "flex", alignItems: "center", gap: 4, fontSize: 12 }}>
   344→                      📷 Instagram Greeting
   345→                    </label>
   346→                    <textarea
   347→                      className="form-input form-textarea"
   348→                      value={instagramGreeting}
   349→                      onChange={(e) => setInstagramGreeting(e.target.value)}
   350→                      rows={3}
   351→                      placeholder="Hey {name}! 👋 Welcome to {business_name} on Instagram!"
   352→                      style={{ fontSize: 12 }}
   353→                    />
   354→                  </div>
   355→                  <div className="form-group" style={{ marginBottom: 0 }}>
   356→                    <label className="form-label" style={{ display: "flex", alignItems: "center", gap: 4, fontSize: 12 }}>
   357→                      🌐 Facebook Greeting
   358→                    </label>
   359→                    <textarea
   360→                      className="form-input form-textarea"
   361→                      value={facebookGreeting}
   362→                      onChange={(e) => setFacebookGreeting(e.target.value)}
   363→                      rows={3}
   364→                      placeholder="Hi {name}! Welcome to {business_name}! How can we help?"
   365→                      style={{ fontSize: 12 }}
   366→                    />
   367→                  </div>
   368→                  <div className="form-group" style={{ marginBottom: 0 }}>
   369→                    <label className="form-label" style={{ display: "flex", alignItems: "center", gap: 4, fontSize: 12 }}>
   370→                      📱 WhatsApp Greeting
   371→                    </label>
   372→                    <textarea
   373→                      className="form-input form-textarea"
   374→                      value={whatsappGreeting}
   375→                      onChange={(e) => setWhatsappGreeting(e.target.value)}
   376→                      rows={3}
   377→                      placeholder="Hello {name}! 🙏 Thanks for messaging {business_name} on WhatsApp!"
   378→                      style={{ fontSize: 12 }}
   379→                    />
   380→                  </div>
   381→                </div>
   382→              )}
   383→
   384→              {/* Greeting Preview */}
   385→              <div style={{
   386→                padding: "var(--space-md)", background: "rgba(108, 92, 231, 0.05)",
   387→                border: "1px solid rgba(108, 92, 231, 0.15)", borderRadius: "var(--radius-md)",
   388→              }}>
   389→                <div style={{ fontWeight: 600, fontSize: 11, color: "var(--accent-primary-light)", marginBottom: 6, textTransform: "uppercase" }}>Preview</div>
   390→                <div style={{ fontSize: "var(--font-size-sm)", color: "var(--text-secondary)", lineHeight: 1.6 }}>
   391→                  {autoGreetingMessage
   392→                    .replace(/\{business_name\}/g, account.business_name || "Sellora")
   393→                    .replace(/\{name\}/g, "Ahmed")}
   394→                </div>
   395→                {greetingDelaySeconds > 0 && (
   396→                  <div style={{ fontSize: 10, color: "var(--text-tertiary)", marginTop: 4 }}>
   397→                    ⏱ Sent after {greetingDelaySeconds}s delay
   398→                  </div>
   399→                )}
   400→              </div>
   401→            </>
   402→          )}
   403→        </div>
   404→      </div>
   405→
   406→      {/* Section 2: Auto Follow-Up for Unpaid Orders */}
   407→      <div className="dashboard-panel" style={{ marginBottom: "var(--space-xl)" }}>
   408→        <div className="dashboard-panel-header">
   409→          <h3 style={{ display: "flex", alignItems: "center", gap: 8 }}>
   410→            <Clock size={18} style={{ color: "var(--accent-orange)" }} />
   411→            Auto Follow-Up for Unpaid Orders
   412→          </h3>
   413→        </div>
   414→        <div className="dashboard-panel-body" style={{ padding: "var(--space-xl)" }}>
   415→          <div style={{
   416→            display: "flex", alignItems: "center", justifyContent: "space-between",
   417→            padding: "var(--space-lg)", background: "var(--bg-glass)",
   418→            borderRadius: "var(--radius-md)", marginBottom: "var(--space-lg)",
   419→          }}>
   420→            <div>
   421→              <div style={{ fontWeight: 600, marginBottom: 4 }}>Auto Follow-Up</div>
   422→              <div style={{ fontSize: "var(--font-size-sm)", color: "var(--text-tertiary)" }}>
   423→                Automatically send follow-up messages to customers with unpaid orders after 24 hours
   424→              </div>
   425→            </div>
   426→            <div
   427→              style={{ color: autoFollowUp ? "var(--accent-green)" : "var(--text-tertiary)", cursor: "pointer" }}
   428→              onClick={() => setAutoFollowUp(!autoFollowUp)}
   429→            >
   430→              {autoFollowUp ? <ToggleRight size={36} /> : <ToggleLeft size={36} />}
   431→            </div>
   432→          </div>
   433→
   434→          <div style={{ display: "flex", alignItems: "center", gap: "var(--space-md)" }}>
   435→            <button
   436→              className="btn btn-secondary"
   437→              disabled={sendingFollowUps}
   438→              onClick={handleSendFollowUps}
   439→            >
   440→              {sendingFollowUps ? <><Loader2 size={16} className="spin" /> Sending...</> : <><Clock size={16} /> Send Follow-Ups Now</>}
   441→            </button>
   442→            <p style={{ fontSize: "var(--font-size-xs)", color: "var(--text-tertiary)" }}>
   443→              Manually trigger follow-up messages for all unpaid orders older than 24 hours
   444→            </p>
   445→          </div>
   446→        </div>
   447→      </div>
   448→
   449→      {/* Section 2.5: Abandoned Cart Recovery */}
   450→      <div className="dashboard-panel" style={{ marginBottom: "var(--space-xl)" }}>
   451→        <div className="dashboard-panel-header">
   452→          <h3 style={{ display: "flex", alignItems: "center", gap: 8 }}>
   453→            <ShoppingCart size={18} style={{ color: "var(--accent-orange)" }} />
   454→            Abandoned Cart Recovery
   455→          </h3>
   456→          <Link
   457→            href="/dashboard/abandoned-carts"
   458→            style={{
   459→              display: "inline-flex", alignItems: "center", gap: 6,
   460→              color: "var(--accent-primary)", fontWeight: 500,
   461→              fontSize: "var(--font-size-sm)", textDecoration: "none",
   462→            }}
   463→          >
   464→            View Carts <ChevronRight size={14} />
   465→          </Link>
   466→        </div>
   467→        <div className="dashboard-panel-body" style={{ padding: "var(--space-xl)" }}>
   468→          {/* Main toggle */}
   469→          <div style={{
   470→            display: "flex", alignItems: "center", justifyContent: "space-between",
   471→            padding: "var(--space-lg)", background: "var(--bg-glass)",
   472→            borderRadius: "var(--radius-md)", marginBottom: "var(--space-lg)",
   473→          }}>
