    17→function TrendArrow({ value }) {
    18→  if (value == null || value === 0) return null;
    19→  const isUp = value > 0;
    20→  return (
    21→    <span className={`stat-card-trend ${isUp ? "up" : "down"}`}>
    22→      {isUp ? <ArrowUpRight size={12} /> : <ArrowDownRight size={12} />}
    23→      {Math.abs(value)}%
    24→    </span>
    25→  );
    26→}
