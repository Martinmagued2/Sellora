    23→    23→function ParticleCanvas() {
    24→    24→  const canvasRef = useRef(null);
    25→    25→  const mouse = useRef({ x: -1000, y: -1000 });
    26→    26→  const particles = useRef([]);
    27→    27→  const rafId = useRef(null);
    28→    28→
    29→    29→  useEffect(() => {
    30→    30→    const canvas = canvasRef.current;
    31→    31→    if (!canvas) return;
    32→    32→    const ctx = canvas.getContext("2d");
    33→    33→    let w, h;
    34→    34→
    35→    35→    function resize() { w = canvas.width = window.innerWidth; h = canvas.height = window.innerHeight; }
    36→    36→    resize();
    37→    37→    window.addEventListener("resize", resize);
    38→    38→
    39→    39→    const count = Math.min(80, Math.floor(window.innerWidth / 18));
    40→    40→    particles.current = Array.from({ length: count }, () => ({
    41→    41→      x: Math.random() * w, y: Math.random() * h,
    42→    42→      vx: (Math.random() - 0.5) * 0.4, vy: (Math.random() - 0.5) * 0.4,
    43→    43→      r: Math.random() * 2 + 1,
    44→    44→      color: Math.random() > 0.5 ? "88,101,242" : "0,210,255",
    45→    45→    }));
    46→    46→
    47→    47→    function draw() {
    48→    48→      ctx.clearRect(0, 0, w, h);
    49→    49→      const ps = particles.current;
    50→    50→      const mx = mouse.current.x; const my = mouse.current.y;
    51→    51→      for (let i = 0; i < ps.length; i++) {
    52→    52→        const p = ps[i];
    53→    53→        const dx = p.x - mx; const dy = p.y - my;
    54→    54→        const dist = Math.sqrt(dx * dx + dy * dy);
    55→    55→        if (dist < 180) { const force = (180 - dist) / 180 * 0.8; p.vx += (dx / dist) * force; p.vy += (dy / dist) * force; }
    56→    56→        p.vx *= 0.98; p.vy *= 0.98; p.x += p.vx; p.y += p.vy;
    57→    57→        if (p.x < 0) p.x = w; if (p.x > w) p.x = 0; if (p.y < 0) p.y = h; if (p.y > h) p.y = 0;
    58→    58→        ctx.beginPath(); ctx.arc(p.x, p.y, p.r, 0, Math.PI * 2);
    59→    59→        ctx.fillStyle = `rgba(${p.color},0.6)`; ctx.fill();
    60→    60→        for (let j = i + 1; j < ps.length; j++) {
    61→    61→          const p2 = ps[j]; const d = Math.hypot(p.x - p2.x, p.y - p2.y);
    62→    62→          if (d < 140) { ctx.beginPath(); ctx.moveTo(p.x, p.y); ctx.lineTo(p2.x, p2.y); ctx.strokeStyle = `rgba(${p.color},${0.15 * (1 - d / 140)})`; ctx.lineWidth = 0.5; ctx.stroke(); }
    63→    63→        }
    64→    64→      }
    65→    65→      rafId.current = requestAnimationFrame(draw);
    66→    66→    }
    67→    67→    draw();
    68→    68→    const handleMouse = (e) => { mouse.current = { x: e.clientX, y: e.clientY }; };
    69→    69→    const handleLeave = () => { mouse.current = { x: -1000, y: -1000 }; };
    70→    70→    window.addEventListener("mousemove", handleMouse); window.addEventListener("mouseleave", handleLeave);
    71→    71→    return () => { cancelAnimationFrame(rafId.current); window.removeEventListener("resize", resize); window.removeEventListener("mousemove", handleMouse); window.removeEventListener("mouseleave", handleLeave); };
    72→    72→  }, []);
    73→    73→
    74→    74→  return <canvas ref={canvasRef} style={{ position: "fixed", inset: 0, zIndex: 0, pointerEvents: "none" }} />;
    75→    75→}

   103→   103→function AnimatedCounter({ value, duration = 1.5 }) {
   104→   104→  const numericValue = typeof value === "number" ? value : parseFloat(value);
   105→   105→  const isNumeric = !isNaN(numericValue);
   106→   106→  const [displayValue, setDisplayValue] = useState(isNumeric ? 0 : value);
   107→   107→  const ref = useRef(null);
   108→   108→  const isInView = useInView(ref, { once: true, margin: "-50px" });
   109→   109→
   110→   110→  useEffect(() => {
   111→   111→    if (!isInView || !isNumeric) return;
   112→   112→    let start = 0; const end = numericValue; const increment = end / (duration * 60);
   113→   113→    const timer = setInterval(() => { start += increment; if (start >= end) { setDisplayValue(end); clearInterval(timer); } else { setDisplayValue(Math.floor(start)); } }, 1000 / 60);
   114→   114→    return () => clearInterval(timer);
   115→   115→  }, [isInView, numericValue, isNumeric, duration]);
   116→   116→
   117→   117→  return <span ref={ref}>{isNumeric ? Math.round(displayValue).toLocaleString() : displayValue}</span>;
   118→   118→}

   123→   123→function TiltCard({ children, className = "", style = {} }) {
   124→   124→  const cardRef = useRef(null);
   125→   125→  const [transform, setTransform] = useState("perspective(1000px) rotateX(0deg) rotateY(0deg)");
   126→   126→
   127→   127→  const handleMouseMove = useCallback((e) => {
   128→   128→    if (!cardRef.current) return;
   129→   129→    const rect = cardRef.current.getBoundingClientRect();
   130→   130→    const x = e.clientX - rect.left;
   131→   131→    const y = e.clientY - rect.top;
   132→   132→    const centerX = rect.width / 2;
   133→   133→    const centerY = rect.height / 2;
   134→   134→    const rotateX = ((y - centerY) / centerY) * -8;
   135→   135→    const rotateY = ((x - centerX) / centerX) * 8;
   136→   136→    setTransform(`perspective(1000px) rotateX(${rotateX}deg) rotateY(${rotateY}deg) scale3d(1.02,1.02,1.02)`);
   137→   137→  }, []);
   138→   138→
   139→   139→  const handleMouseLeave = useCallback(() => {
   140→   140→    setTransform("perspective(1000px) rotateX(0deg) rotateY(0deg) scale3d(1,1,1)");
   141→   141→  }, []);
   142→   142→
   143→   143→  return (
   144→   144→    <div
   145→   145→      ref={cardRef}
   146→   146→      className={`tilt-card ${className}`}
   147→   147→      style={{ ...style, transform, transition: "transform 0.15s ease-out", transformStyle: "preserve-3d" }}
   148→   148→      onMouseMove={handleMouseMove}
   149→   149→      onMouseLeave={handleMouseLeave}
   150→   150→    >
   151→   151→      {children}
   152→   152→    </div>
   153→   153→  );
   154→   154→}

   483→   483→   SVG HEX GRID — subtle decorative hexagons
   484→   484→   ============================================ */
   485→   485→function SVGHexGrid() {
   486→   486→  const hexSize = 30;
   487→   487→  const cols = 8;
   488→   488→  const rows = 4;
   489→   489→  const hexW = hexSize * 2;
   490→   490→  const hexH = Math.sqrt(3) * hexSize;
   491→   491→
   492→   492→  const hexPoints = (cx, cy, r) => {
   493→   493→    return Array.from({ length: 6 }, (_, i) => {
   494→   494→      const angle = (Math.PI / 180) * (60 * i - 30);
   495→   495→      return `${cx + r * Math.cos(angle)},${cy + r * Math.sin(angle)}`;
   496→   496→    }).join(" ");
   497→   497→  };
   498→   498→
   499→   499→  return (
   500→   500→    <svg className="svg-hex-grid" viewBox="0 0 500 250" preserveAspectRatio="xMidYMid slice" aria-hidden="true">
   501→   501→      {Array.from({ length: rows }, (_, row) =>
   502→   502→        Array.from({ length: cols }, (_, col) => {
   503→   503→          const cx = col * hexW * 0.75 + 30;
   504→   504→          const cy = row * hexH + (col % 2 === 1 ? hexH / 2 : 0) + 30;
   505→   505→          const opacity = 0.03 + Math.random() * 0.04;
   506→   506→          return (
   507→   507→            <polygon
   508→   508→              key={`${row}-${col}`}
   509→   509→              points={hexPoints(cx, cy, hexSize - 2)}
   510→   510→              fill="none"
   511→   511→              stroke="var(--accent-primary)"
   512→   512→              strokeWidth="0.5"
   513→   513→              strokeOpacity={opacity}
   514→   514→            />
   515→   515→          );
   516→   516→        })
   517→   517→      )}
   518→   518→    </svg>
   519→   519→  );
   520→   520→}
