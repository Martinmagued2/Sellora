     6→let _supabase = null;
     7→function getSupabase() {
     8→  if (!_supabase) {
     9→    _supabase = createClient(
    10→      process.env.NEXT_PUBLIC_SUPABASE_URL,
    11→      process.env.SUPABASE_SERVICE_ROLE_KEY
    12→    );
    13→  }
    14→  return _supabase;
    15→}

    17→// Helper to fetch account context
    18→const getAccountCurrency = async (accountId) => {
    19→  const { data } = await getSupabase()
    20→    .from("accounts")
    21→    .select("currency")
    22→    .eq("id", accountId)
    23→    .single();
    24→  return data?.currency || "EGP";
    25→};

    99→                    if (pidArray.includes(item.product_id)) {

Searches product names, descriptions, and categories to find the best matches.",
   187→      inputSchema: z.object({
   188→        query: z.string().describe("The customer's need, preference, or context (e.g. 'dry skin', 'gift for mom', 'casual outfit')"),
   189→      }),
   190→      execute: async ({ query }) => {
   191→        if (!query) {
   192→          return { success: false, error: "Please describe what you're looking for" };
   193→        }
   194→
   195→        const searchTerms = query.toLowerCase().split(/\s+/).filter(Boolean);
   196→
   197→        // Fetch all active products for this account (exclude hidden_from_ai and out-of-stock)
   198→        const { data: products, error } = await getSupabase()
   199→          .from("products")
   200→          .select("id, name, description, price, stock, category, variants")
   201→          .eq("account_id", accountId)

   203→          .neq("hidden_from_ai", true)

   207→          return { success: false, error: "Failed to search products" };
   208→        }
   209→
   210→        if (!products || products.length === 0) {
   211→          return { success: true, message: "No products available at the moment.", products: [] };
   212→        }
   213→
   214→        // Score each product based on relevance to the query
   215→        const scored = products.map((product) => {
   216→          let score = 0;
   217→          const nameLower = (product.name || "").toLowerCase();
   218→          const descLower = (product.description || "").toLowerCase();
   219→          const catLower = (product.category || "").toLowerCase();
   220→          // Also search in variant names (e.g. "red", "large", "32GB")
   221→          const variantText = (product.variants || []).map(v => v.name).join(" ").toLowerCase();
   222→          const allText = `${nameLower} ${descLower} ${catLower} ${variantText}`;
   223→
   224→          for (const term of searchTerms) {
   225→            if (nameLower.includes(term)) score += 10; // Name match is strongest
   226→            if (catLower.includes(term)) score += 8;   // Category match
   227→            if (descLower.includes(term)) score += 5;   // Description match
   228→            if (variantText.includes(term)) score += 7;  // Variant match (e.g. "red", "large")
   229→            if (allText.includes(term)) score += 2;     // Any mention
   230→          }
   231→
   232→          // Boost products that are in stock
   233→          if (product.stock > 0) score += 3;
   234→
   235→          return { ...product, score };
   236→        });
   237→
   238→        // Filter to products with some relevance and sort by score
   239→        const recommendations = scored
   240→          .filter((p) => p.score > 0)
   241→          .sort((a, b) => b.score - a.score)
   242→          .slice(0, 5);
   243→
   244→        if (recommendations.length === 0) {
   245→          return { success: true, message: `I couldn't find products matching "${query}". Let me show you what we have available.`, products: products.slice(0, 5) };
   246→        }
   247→
   248→        const currency = await getAccountCurrency(accountId);
   249→        return { success: true, currency, query, products: recommendations.map(({ score, ...rest }) => rest) };
   250→      },
