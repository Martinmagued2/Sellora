   526→      // Send confirmation via Meta API for Facebook/Instagram channels or WhatsApp

   528→        if (activeConv.channel === "instagram" || activeConv.channel === "facebook" || activeConv.channel === "whatsapp") {

   530→            method: "POST",
   531→            headers: { "Content-Type": "application/json" },
   532→            body: JSON.stringify({
   533→              conversationId: activeConv.id,
   534→              content: confirmationContent,
   535→              type: "text",
   536→              ...(activeConv.channel === "whatsapp" && { channel: "whatsapp" }),
