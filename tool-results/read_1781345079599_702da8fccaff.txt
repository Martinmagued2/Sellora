    26→
    27→export default function ProductsPage() {
    28→  const router = useRouter();
    29→  const toast = useToast();
    30→  
    31→  const confirmAction = useConfirm();
    32→  const [products, setProducts] = useState([]);
    33→  const [filter, setFilter] = useState("all");
    34→  const [search, setSearch] = useState("");
    35→  const [showModal, setShowModal] = useState(false);
    36→  const [loading, setLoading] = useState(true);
    37→  const [saving, setSaving] = useState(false);
    38→  const [editingProduct, setEditingProduct] = useState(null);
    39→  const [imageFile, setImageFile] = useState(null);
    40→  const [imagePreview, setImagePreview] = useState(null);
    41→  const [dragOver, setDragOver] = useState(false);
    42→  const [accountPlan, setAccountPlan] = useState("starter");
    43→  const [generatingImage, setGeneratingImage] = useState(false);
    44→  const [aiStyle, setAiStyle] = useState("studio");
    45→  const [generatedImageUrl, setGeneratedImageUrl] = useState(null);
    46→  const [generatingDesc, setGeneratingDesc] = useState(false);
    47→  const [aiDescEnglish, setAiDescEnglish] = useState("");
    48→  const [aiDescArabic, setAiDescArabic] = useState("");
    49→  const [aiPriceSuggestion, setAiPriceSuggestion] = useState("");
    50→  const [topRecommended, setTopRecommended] = useState([]);
    51→  const [recLoading, setRecLoading] = useState(false);
    52→
    53→  // Variants state
    54→  const [variants, setVariants] = useState([]);
    55→
    56→  // Active coupons for discount display
    57→  const [activeCoupons, setActiveCoupons] = useState([]);
    58→
    59→  // View product details modal
    60→  const [viewProduct, setViewProduct] = useState(null);
    61→
    62→  const fileInputRef = useRef(null);
    63→
    64→  const { currentStoreId } = useCurrentStore();
    65→
    66→  const supabase = createClient();
    67→  const planLimits = getPlanLimits(accountPlan);
    68→  const limitReached = planLimits.products !== -1 && products.length >= planLimits.products;
    69→
    70→  // Ensure storage buckets exist
    71→  const ensureBuckets = useCallback(async () => {
    72→    try {
    73→      await fetch("/api/storage/ensure-buckets", { method: "POST" });
    74→    } catch (e) {
    75→      console.warn("[Products] Bucket ensure failed:", e.message);
    76→    }
    77→  }, []);
    78→
    79→  const fetchProducts = useCallback(async () => {
    80→    setLoading(true);
    81→
    82→    const { data: { user } } = await supabase.auth.getUser();
    83→    if (user) {
    84→      const { data: account } = await supabase.from("accounts").select("plan").eq("id", user.id).single();
    85→      if (account?.plan) setAccountPlan(account.plan);
    86→    }
    87→    if (!user) { setLoading(false); return; }
    88→
    89→    let query = supabase.from("products").select("*").eq("account_id", user.id).order("created_at", { ascending: false });
    90→
    91→    if (filter === "active") query = query.eq("status", "active");
    92→    if (filter === "draft") query = query.eq("status", "draft");
    93→    if (filter === "low") query = query.lte("stock", 5).gt("stock", 0);
    94→    if (search) query = query.ilike("name", `%${search}%`);
    95→    if (currentStoreId) query = query.or(`store_id.eq.${currentStoreId},store_id.is.null`);
    96→
    97→    const { data, error } = await query;
    98→    if (!error) setProducts(data || []);
    99→    setLoading(false);
   100→  }, [filter, search, currentStoreId]);
   101→
   102→  useEffect(() => { ensureBuckets(); fetchProducts(); }, [ensureBuckets, fetchProducts]);
   103→
   104→  // Fetch active coupons for discount display
   105→  useEffect(() => {
   106→    const fetchActiveCoupons = async () => {
   107→      try {
   108→        const { data: { user } } = await supabase.auth.getUser();
   109→        if (!user) return;
   110→        const now = new Date().toISOString();
   111→        const { data, error } = await supabase
   112→          .from("coupons")
   113→          .select("*")
   114→          .eq("account_id", user.id)
   115→          .eq("is_active", true)
   116→          .lte("starts_at", now)
   117→          .or(`expires_at.is.null,expires_at.gte.${now}`);
   118→        if (!error && data) {
   119→          // Filter out exhausted coupons
   120→          const active = data.filter(c => c.max_uses === null || c.used_count < c.max_uses);
   121→          setActiveCoupons(active);
   122→        }
   123→      } catch (err) {
   124→        console.warn("Failed to fetch active coupons:", err.message);
   125→      }
   126→    };
   127→    fetchActiveCoupons();
   128→  }, [supabase]);
   129→
   130→  // Helper: find the best applicable coupon for a product
   131→  const getDiscountForProduct = useCallback((product) => {
   132→    if (!activeCoupons.length) return null;
   133→    let bestDiscount = null;
   134→    let bestSavings = 0;
   135→
   136→    for (const coupon of activeCoupons) {
   137→      // Check if coupon applies to this product
   138→      let applies = false;
   139→      if (coupon.applies_to === "all") {
   140→        applies = true;
   141→      } else if (coupon.applies_to === "specific_products") {
   142→        applies = Array.isArray(coupon.product_ids) && coupon.product_ids.includes(product.id);
   143→      } else if (coupon.applies_to === "specific_categories") {
   144→        applies = Array.isArray(coupon.categories) && product.category && coupon.categories.includes(product.category);
   145→      }
   146→
   147→      if (!applies) continue;
   148→
   149→      // Check min_order_value (product price as proxy)
   150→      if (coupon.min_order_value && product.price < coupon.min_order_value) continue;
   151→
   152→      // Calculate savings
   153→      let savings = 0;
   154→      if (coupon.type === "percentage") {
   155→        savings = product.price * (coupon.value / 100);
   156→      } else if (coupon.type === "fixed") {
   157→        savings = Math.min(parseFloat(coupon.value), product.price);
   158→      } else if (coupon.type === "free_shipping") {
   159→        // Free shipping doesn't change product price display
   160→        continue;
   161→      }
   162→
   163→      // Pick the coupon that gives the biggest discount
   164→      if (savings > bestSavings) {
   165→        bestSavings = savings;
   166→        bestDiscount = {
   167→          coupon,
   168→          discountAmount: savings,
   169→          newPrice: Math.max(product.price - savings, 0),
   170→          label: coupon.type === "percentage" ? `${coupon.value}% OFF` : `${parseFloat(coupon.value).toLocaleString()} EGP OFF`,
   171→        };
   172→      }
   173→    }
   174→
   175→    return bestDiscount;
   176→  }, [activeCoupons]);
   177→
   178→  // Fetch top recommended products
   179→  const fetchTopRecommended = useCallback(async () => {
   180→    setRecLoading(true);
   181→    try {
   182→      const { data: { user } } = await supabase.auth.getUser();
   183→      if (!user) return;
   184→
   185→      const { data: orders } = await supabase
   186→        .from("orders")
   187→        .select("items")
   188→        .eq("account_id", user.id)
   189→        .limit(500);
   190→
   191→      if (!orders || orders.length === 0) { setTopRecommended([]); setRecLoading(false); return; }
   192→
   193→      const productCounts = {};
   194→      for (const order of orders) {
   195→        if (order.items && Array.isArray(order.items)) {
   196→          for (const item of order.items) {
   197→            if (item.product_id) {
   198→              productCounts[item.product_id] = (productCounts[item.product_id] || 0) + 1;
   199→            }
   200→          }
   201→        }
   202→      }
   203→
   204→      const topIds = Object.entries(productCounts)
   205→        .sort((a, b) => b[1] - a[1])
   206→        .slice(0, 5)
   207→        .map(([id, count]) => ({ id, count }));
   208→
   209→      if (topIds.length === 0) { setTopRecommended([]); setRecLoading(false); return; }
   210→
   211→      const { data: prods } = await supabase
   212→        .from("products")
   213→        .select("id, name, price, category, image_urls, stock")
   214→        .in("id", topIds.map(t => t.id));
   215→
   216→      const merged = topIds.map(t => {
   217→        const prod = prods?.find(p => p.id === t.id);
   218→        return prod ? { ...prod, rec_count: t.count } : null;
   219→      }).filter(Boolean);
   220→
   221→      setTopRecommended(merged);
   222→    } catch (err) {
   223→      console.error("Failed to fetch top recommended:", err);
   224→    }
   225→    setRecLoading(false);
   226→  }, [supabase]);
   227→
   228→  useEffect(() => { fetchTopRecommended(); }, [fetchTopRecommended]);
   229→
   230→  // ─── Variant helpers ───
   231→  const addVariant = () => {
   232→    setVariants((prev) => [
   233→      ...prev,
   234→      { name: "", sku: "", price: 0, stock: 0, image_url: "" },
   235→    ]);
   236→  };
   237→
   238→  const removeVariant = (index) => {
   239→    setVariants((prev) => prev.filter((_, i) => i !== index));
   240→  };
   241→
   242→  const updateVariant = (index, field, value) => {
   243→    setVariants((prev) => {
   244→      const copy = [...prev];
   245→      copy[index] = { ...copy[index], [field]: value };
   246→      return copy;
   247→    });
   248→  };
   249→
   250→  // ─── File handling ───
   251→  const handleFileSelect = (file) => {
   252→    if (!file) return;
   253→    if (!file.type.startsWith("image/")) {
   254→      toast.warning("Please select an image file (jpg, png, webp)");
   255→      return;
   256→    }
   257→    if (file.size > 5 * 1024 * 1024) {
   258→      toast.warning("Image must be smaller than 5MB");
   259→      return;
   260→    }
   261→    setImageFile(file);
   262→    setGeneratedImageUrl(null);
   263→    const reader = new FileReader();
   264→    reader.onloadend = () => setImagePreview(reader.result);
   265→    reader.readAsDataURL(file);
   266→  };
   267→
   268→  const handleDragOver = (e) => { e.preventDefault(); setDragOver(true); };
   269→  const handleDragLeave = (e) => { e.preventDefault(); setDragOver(false); };
   270→  const handleDrop = (e) => {
   271→    e.preventDefault();
   272→    setDragOver(false);
   273→    handleFileSelect(e.dataTransfer.files[0]);
   274→  };
   275→
   276→  const handleGenerateImage = async () => {
   277→    const form = document.querySelector('.modal form');
   278→    const productName = form?.elements?.name?.value;
   279→    const description = form?.elements?.description?.value;
   280→
   281→    if (!productName?.trim()) {
   282→      toast.warning("Please enter a product name first, then generate an image.");
   283→      return;
   284→    }
   285→
   286→    setGeneratingImage(true);
   287→    try {
   288→      const res = await fetch("/api/products/generate-image", {
   289→        method: "POST",
   290→        headers: { "Content-Type": "application/json" },
   291→        body: JSON.stringify({
   292→          product_name: productName.trim(),
   293→          description: description?.trim() || "",
   294→          style: aiStyle,
   295→        }),
   296→      });
   297→
   298→      const data = await res.json();
   299→      if (!res.ok || !data.success) {
   300→        toast.error(data.error || "Image generation failed. Please try again.");
   301→        return;
   302→      }
   303→
   304→      const dataUrl = `data:image/png;base64,${data.image_base64}`;
   305→      setImagePreview(dataUrl);
   306→      setImageFile(null);
   307→      setGeneratedImageUrl(data.image_url);
   308→    } catch (err) {
   309→      console.error("Image generation error:", err);
   310→      toast.error("Image generation failed. Please try again.");
   311→    } finally {
   312→      setGeneratingImage(false);
   313→    }
   314→  };
   315→
   316→  const handleAddProduct = async (e) => {
   317→    e.preventDefault();
   318→    setSaving(true);
   319→    const fd = new FormData(e.target);
   320→    const { data: { user } } = await supabase.auth.getUser();
   321→
   322→    let imageUrl = null;
   323→
   324→    if (generatedImageUrl) {
   325→      imageUrl = generatedImageUrl;
   326→    } else if (imageFile) {
   327→      try {
   328→        await fetch("/api/storage/ensure-buckets", { method: "POST" });
   329→      } catch (e) {}
   330→
   331→      const ext = imageFile.name.split(".").pop();
   332→      const fileName = `${user.id}/${Date.now()}.${ext}`;
   333→
   334→      let uploadError;
   335→      let uploadData;
   336→
   337→      const clientResult = await supabase.storage
   338→        .from("product-images")
   339→        .upload(fileName, imageFile, { cacheControl: "3600", upsert: false });
   340→      uploadError = clientResult.error;
   341→      uploadData = clientResult.data;
   342→
   343→      if (uploadError) {
   344→        console.warn("[Products] Client upload failed, trying admin:", uploadError.message);
   345→        try {
   346→          const formData = new FormData();
   347→          formData.append("file", imageFile);
   348→          formData.append("path", fileName);
   349→          const adminRes = await fetch("/api/storage/upload", {
   350→            method: "POST",
   351→            body: formData,
   352→          });
   353→          if (adminRes.ok) {
   354→            const adminData = await adminRes.json();
   355→            if (adminData.url) {
   356→              imageUrl = adminData.url;
   357→              uploadError = null;
   358→            }
   359→          }
   360→        } catch (adminErr) {
   361→          console.error("[Products] Admin upload also failed:", adminErr.message);
   362→        }
   363→      }
   364→
   365→      if (!uploadError && uploadData) {
   366→        const { data: urlData } = supabase.storage
   367→          .from("product-images")
   368→          .getPublicUrl(fileName);
   369→        imageUrl = urlData.publicUrl;
   370→      } else if (uploadError && !imageUrl) {
   371→        toast.error("Image upload failed: " + (uploadError.message || "Unknown error."));
   372→      }
   373→    }
   374→
   375→    // Build variants array: clean up empty entries
   376→    const hasVariants = variants.some((v) => v.name.trim() !== "");
   377→    const cleanVariants = variants
   378→      .filter((v) => v.name.trim() !== "")
   379→      .map((v) => ({
   380→        name: v.name.trim(),
   381→        sku: v.sku.trim() || null,
   382→        price: Number(v.price) || 0,
   383→        stock: Number(v.stock) || 0,
   384→        image_url: v.image_url.trim() || null,
   385→      }));
   386→
   387→    // Calculate base price/stock from variants or form fields
   388→    let basePrice, baseStock;
   389→    if (hasVariants) {
   390→      // When variants exist, use the lowest variant price as the base price
   391→      // and sum all variant stocks as the total stock
   392→      basePrice = cleanVariants.length > 0 ? Math.min(...cleanVariants.map(v => v.price)) : 0;
   393→      baseStock = cleanVariants.reduce((sum, v) => sum + v.stock, 0);
   394→    } else {
   395→      basePrice = parseFloat(fd.get("price"));
   396→      baseStock = parseInt(fd.get("stock"));
   397→    }
   398→
   399→    const payload = {
   400→      account_id: user.id,
   401→      name: fd.get("name"),
   402→      description: fd.get("description"),
   403→      price: basePrice,
   404→      category: fd.get("category"),
   405→      stock: baseStock,
   406→      status: fd.get("status") || "active",
   407→      variants: cleanVariants,
   408→    };
   409→
   410→    if (imageUrl) {
   411→      payload.image_urls = [imageUrl];
   412→    } else if (editingProduct && !imagePreview) {
   413→      payload.image_urls = [];
   414→    }
   415→
   416→    let error;
   417→    if (editingProduct) {
   418→      const { error: updateError } = await supabase.from("products").update(payload).eq("id", editingProduct.id).eq("account_id", user.id);
   419→      error = updateError;
   420→    } else {
   421→      if (!imageUrl && !editingProduct) {
   422→        payload.image_urls = [];
   423→      }
   424→      const { error: insertError } = await supabase.from("products").insert(payload);
   425→      error = insertError;
   426→    }
   427→
   428→    if (!error) {
   429→      closeModal();
   430→      fetchProducts();
   431→    }
   432→    setSaving(false);
   433→  };
   434→
   435→  const handleEdit = async (product) => {
   436→    setEditingProduct(product);
   437→    setImagePreview(product.image_urls && product.image_urls.length > 0 ? product.image_urls[0] : null);
   438→    // Load existing variants
   439→    const existingVariants = Array.isArray(product.variants) ? product.variants : [];
   440→    setVariants(
   441→      existingVariants.map((v) => ({
   442→        name: v.name || "",
   443→        sku: v.sku || "",
   444→        price: v.price ?? v.price_offset ?? 0,
   445→        stock: v.stock ?? 0,
   446→        image_url: v.image_url || "",
   447→      }))
   448→    );
   449→    setShowModal(true);
   450→  };
   451→
   452→  const handleDelete = async (id, imageUrls) => {
   453→    if (!(await confirmAction("Delete this product?"))) return;
   454→
   455→    if (imageUrls && imageUrls.length > 0) {
   456→      try {
   457→        const { data: { user } } = await supabase.auth.getUser();
   458→        if (user) {
   459→          const pathsToRemove = imageUrls.map(url => {
   460→            const parts = url.split("product-images/");
   461→            return parts.length > 1 ? parts[1] : null;
   462→          }).filter(Boolean);
   463→
   464→          if (pathsToRemove.length > 0) {
   465→            await supabase.storage.from("product-images").remove(pathsToRemove);
   466→          }
   467→        }
   468→      } catch (err) {
   469→        console.error("Failed to delete product images from storage:", err);
   470→      }
   471→    }
   472→
   473→    const { data: { user: delUser } } = await supabase.auth.getUser();
   474→    await supabase.from("products").delete().eq("id", id).eq("account_id", delUser?.id);
   475→    fetchProducts();
   476→  };
   477→
   478→  const closeModal = () => {
   479→    setShowModal(false);
   480→    setEditingProduct(null);
   481→    setImageFile(null);
   482→    setImagePreview(null);
   483→    setGeneratingImage(false);
   484→    setAiStyle("studio");
   485→    setGeneratedImageUrl(null);
   486→    setGeneratingDesc(false);
   487→    setAiDescEnglish("");
   488→    setAiDescArabic("");
   489→    setAiPriceSuggestion("");
   490→    setVariants([]);
   491→  };
   492→
   493→  const handleGenerateDescription = async () => {
   494→    const form = document.querySelector('.modal form');
   495→    const productName = form?.elements?.name?.value;
   496→    const category = form?.elements?.category?.value;
   497→
   498→    if (!productName?.trim()) {
   499→      toast.warning("Please enter a product name first, then generate a description.");
   500→      return;
   501→    }
   502→
   503→    setGeneratingDesc(true);
   504→    try {
   505→      const res = await fetch("/api/ai/generate-description", {
   506→        method: "POST",
   507→        headers: { "Content-Type": "application/json" },
   508→        body: JSON.stringify({
   509→          product_name: productName.trim(),
   510→          features: productName.trim(),
   511→          category: category || "General",
   512→        }),
   513→      });
   514→
   515→      const data = await res.json();
   516→      if (!res.ok || !data.success) {
   517→        toast.error(data.error || "Description generation failed.");
   518→        return;
   519→      }
   520→
   521→      if (data.english) {
   522→        setAiDescEnglish(data.english);
   523→        const descField = form?.elements?.description;
   524→        if (descField) descField.value = data.english;
   525→      }
   526→      if (data.arabic) setAiDescArabic(data.arabic);
   527→      if (data.price_suggestion) setAiPriceSuggestion(data.price_suggestion);
   528→    } catch (err) {
   529→      console.error("Description generation error:", err);
   530→      toast.error("Description generation failed. Please try again.");
   531→    } finally {
   532→      setGeneratingDesc(false);
   533→    }
   534→  };
   535→
   536→  const emojis = { Bags: "👜", Jewelry: "💎", Accessories: "🧣", Electronics: "📱", Watches: "⌚", Clothing: "👗" };
   537→
   538→  return (
   539→    <>
   540→      <div className="page-header">
   541→        <h1>Products</h1>
   542→        <div className="page-header-actions">
   543→          {limitReached ? (
   544→            <button className="btn btn-primary" onClick={() => router.push('/dashboard/billing')} id="add-product" style={{ opacity: 0.7 }}>
   545→               Upgrade to Add More
   546→            </button>
   547→          ) : (
   548→            <button className="btn btn-primary" onClick={() => { setVariants([]); setShowModal(true); }} id="add-product">
   549→              <Plus size={16} /> Add Product
   550→            </button>
   551→          )}
   552→        </div>
   553→      </div>
   554→
   555→      <div className="filter-bar">
   556→        <div className="filter-tabs">
   557→          {[
   558→            { key: "all", label: "All Products" },
   559→            { key: "active", label: "Active" },
   560→            { key: "draft", label: "Draft" },
   561→            { key: "low", label: "Low Stock" },
   562→          ].map((f) => (
   563→            <button key={f.key} className={`filter-tab ${filter === f.key ? "active" : ""}`} onClick={() => setFilter(f.key)}>
   564→              {f.label}
   565→            </button>
   566→          ))}
   567→        </div>
   568→        <div className="filter-search">
   569→          <Search size={14} />
   570→          <input type="text" placeholder="Search products..." value={search} onChange={(e) => setSearch(e.target.value)} />
   571→        </div>
   572→      </div>
   573→
   574→      {loading ? (
   575→        <PageSkeleton showStats={false} showTable={false} />
